#Const Version      "2025-04-22"
#Const ScriptName   "Libs/domino54/HttpClient.Script.txt"

#Include "TextLib" as TL

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_LibHttpClient_DefaultTimeout         5000
#Const C_LibHttpClient_DefaultUseCache        False
#Const C_LibHttpClient_DefaultHeaders         ""
#Const C_LibHttpClient_EHttpMethodGet         "GET"
#Const C_LibHttpClient_EHttpMethodPost        "POST"
#Const C_LibHttpClient_EHttpMethodPostFile    "POSTFILE"
#Const C_LibHttpClient_TimeoutStatusCode      0
#Const C_LibHttpClient_CancelledStatusCode    0

// ---------------------------------- //
// Structs
// ---------------------------------- //

#Struct K_HttpRequestOptions {
  Text Namespace;
  Text Id;
  Text Method;
  Text Url;
  Text Payload;
  Integer Timeout;
  Text Headers;
  Boolean UseCache;
}

#Struct K_HttpClientEvent {
  Text Namespace;
  Text Id;
  Text Url;
  Integer StatusCode;
  Text ResponseBody;
}

// ---------------------------------- //
// Global variables
// ---------------------------------- //

declare Text[] G_LibHttpClient_PendingNamespaces;
declare K_HttpRequestOptions[] G_LibHttpClient_RequestsQueue;
declare K_HttpClientEvent[] G_LibHttpClient_PendingEvents;

// ---------------------------------- //
// Private functions
// ---------------------------------- //

Text Private_GetNamespace(Text _Id) {
  declare Namespace = TL::Split(":", _Id)[0];

  assert(Namespace != _Id, "All requests must use a namespaced id, for example: \"my_lib:player_data\".");

  return Namespace;
}

Void Private_HttpCreate(K_HttpRequestOptions _Options) {
  declare CHttpRequest Request;

  switch (_Options.Method) {
    case C_LibHttpClient_EHttpMethodGet: {
      Request = Http.CreateGet(_Options.Url, _Options.UseCache, _Options.Headers);
    }
    case C_LibHttpClient_EHttpMethodPost: {
      Request = Http.CreatePost(_Options.Url, _Options.Payload, _Options.Headers);
    }
    case C_LibHttpClient_EHttpMethodPostFile: {
      Request = Http.CreatePostFile(_Options.Url, _Options.Payload, _Options.Headers);
    }
    default: {
      assert(False, """Unsupported HTTP request method {{{_Options.Method}}}, supported methods: {{{C_LibHttpClient_EHttpMethodGet}}}, {{{C_LibHttpClient_EHttpMethodPost}}}, {{{C_LibHttpClient_EHttpMethodPostFile}}}""");
    }
  }

  // Requests to local files and cached requests may complete immediately.
  if (Request.IsCompleted) {
    G_LibHttpClient_PendingEvents.add(K_HttpClientEvent{
      Namespace = _Options.Namespace,
      Id = _Options.Id,
      Url = Request.Url,
      StatusCode = Request.StatusCode,
      ResponseBody = Request.Result
    });

    declare Removed = G_LibHttpClient_PendingNamespaces.remove(_Options.Namespace);

    Http.Destroy(Request);

    return;
  }

  declare Text LibHttpClient_Namespace for Request;
  declare Text LibHttpClient_PrivateId for Request;
  declare Integer LibHttpClient_Timeout for Request;

  LibHttpClient_Namespace = _Options.Namespace;
  LibHttpClient_PrivateId = _Options.Id;
  LibHttpClient_Timeout = Now + _Options.Timeout;
}

Void Private_CreateOrEnqueue(K_HttpRequestOptions _Options) {
  G_LibHttpClient_PendingNamespaces.add(_Options.Namespace);

  if (Http.SlotsAvailable > 0) {
    Private_HttpCreate(_Options);
  } else {
    G_LibHttpClient_RequestsQueue.add(_Options);
  }
}

/**
 * This method should be called at the end of your script's execution context.
 * For game modes it can be done inside the `***MB_Private_UnloadLibraries***` label.
 */
Void Unload() {
  G_LibHttpClient_RequestsQueue.clear();
  G_LibHttpClient_PendingEvents.clear();

  declare Ident[] RequestsToDestroy;

  foreach (Request in Http.Requests) {
    declare Text LibHttpClient_Namespace for Request;

    if (LibHttpClient_Namespace != "") {
      RequestsToDestroy.add(Request.Id);
    }
  }

  foreach (RequestId in RequestsToDestroy) {
    Http.Destroy(Http.Requests[RequestId]);
  }
}

/**
 * This method should be called at the beginning of your script's execution context.
 * For game modes it can be done inside the `***MB_Private_LoadLibraries***` label.
 */
Void Load() {
  Unload();
}

/**
 * This method MUST be called after every `yield;` keyword in your script's execution context.
 * This can be done for example in your game mode's `***MB_Private_Yield***` label.
 */
Void Yield() {
  // Enqueue requests that originally have exceeded the available slots.
  while (G_LibHttpClient_RequestsQueue.count > 0 && Http.SlotsAvailable > 0) {
    declare Request = G_LibHttpClient_RequestsQueue[0];
    declare Removed = G_LibHttpClient_RequestsQueue.removekey(0);

    Private_HttpCreate(Request);
  }

  // Handle completed requests.
  foreach (Event in Http.PendingEvents) {
    if (Event.Type != CHttpEvent::EType::RequestComplete || Event.Request == Null) {
      continue;
    }

    declare Text LibHttpClient_Namespace for Event.Request;

    if (LibHttpClient_Namespace == "") {
      continue;
    }

    declare Text LibHttpClient_PrivateId for Event.Request;

    G_LibHttpClient_PendingEvents.add(K_HttpClientEvent{
      Namespace = LibHttpClient_Namespace,
      Id = LibHttpClient_PrivateId,
      Url = Event.Request.Url,
      StatusCode = Event.Request.StatusCode,
      ResponseBody = Event.Request.Result
    });

    declare Removed = G_LibHttpClient_PendingNamespaces.remove(LibHttpClient_Namespace);

    Http.Destroy(Event.Request);
  }

  // Handle custom timeouts.
  if (Http.Requests.count > 0) {
    declare Ident[] RequestsToDestroy;

    foreach (Request in Http.Requests) {
      declare Integer LibHttpClient_Timeout for Request;

      if (Request.IsCompleted || LibHttpClient_Timeout <= 0 || Now < LibHttpClient_Timeout) {
        continue;
      }

      declare Text LibHttpClient_Namespace for Request;
      declare Text LibHttpClient_PrivateId for Request;

      G_LibHttpClient_PendingEvents.add(K_HttpClientEvent{
        Namespace = LibHttpClient_Namespace,
        Id = LibHttpClient_PrivateId,
        Url = Request.Url,
        StatusCode = C_LibHttpClient_TimeoutStatusCode
      });

      declare Removed = G_LibHttpClient_PendingNamespaces.remove(LibHttpClient_Namespace);

      RequestsToDestroy.add(Request.Id);
    }

    foreach (RequestId in RequestsToDestroy) {
      Http.Destroy(Http.Requests[RequestId]);
    }
  }
}

Integer SlotsAvailable() {
  return Http.SlotsAvailable - G_LibHttpClient_RequestsQueue.count;
}

Boolean IsValidUrl(Text _Url) {
  return Http.IsValidUrl(_Url);
}

Boolean HasPendingRequests(Text _Namespace) {
  return G_LibHttpClient_PendingNamespaces.exists(_Namespace);
}

Boolean HasPendingEvents(Text _Namespace) {
  foreach (Event in G_LibHttpClient_PendingEvents) {
    if (Event.Namespace == _Namespace) {
      return True;
    }
  }

  return False;
}

K_HttpClientEvent GetFirstEvent(Text _Namespace) {
  declare Index = -1;
  declare FirstEvent = K_HttpClientEvent{};

  foreach (I => Event in G_LibHttpClient_PendingEvents) {
    if (Event.Namespace == _Namespace) {
      FirstEvent = Event;
      Index = I;
      break;
    }
  }

  if (Index >= 0) {
    declare Removed = G_LibHttpClient_PendingEvents.removekey(Index);
  }

  return FirstEvent;
}

Boolean Cancel(CHttpRequest _Request) {
  if (_Request == Null) {
    return False;
  }

  declare Text LibHttpClient_Namespace for _Request;
  declare Text LibHttpClient_PrivateId for _Request;

  G_LibHttpClient_PendingEvents.add(K_HttpClientEvent{
    Namespace = LibHttpClient_Namespace,
    Id = LibHttpClient_PrivateId,
    Url = _Request.Url,
    StatusCode = C_LibHttpClient_CancelledStatusCode,
    ResponseBody = _Request.Result
  });

  Http.Destroy(_Request);

  return G_LibHttpClient_PendingNamespaces.remove(LibHttpClient_Namespace);
}

Boolean Cancel(Ident _RequestId) {
  if (!Http.Requests.existskey(_RequestId)) {
    return False;
  }

  return Cancel(Http.Requests[_RequestId]);
}
