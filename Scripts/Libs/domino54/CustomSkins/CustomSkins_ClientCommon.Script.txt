/**
 * Set custom skin entry for a model in a given category.
 */
Void SetPlayerSkinId(Text _ModelId, Text _SkinCategoryId, Text _SkinId) {
  if (_ModelId == "") {
    return;
  }

  declare persistent Integer Persistent_LibCustomSkins_LastUpdate for LocalUser;
  declare persistent Text[Text] Persistent_LibCustomSkins_ModelIdToSkinId for LocalUser;

  declare Index = Common::GetIndex(_ModelId, _SkinCategoryId);

  if (_SkinId == "" && Persistent_LibCustomSkins_ModelIdToSkinId.existskey(Index)) {
    declare Removed = Persistent_LibCustomSkins_ModelIdToSkinId.removekey(Index);

    Persistent_LibCustomSkins_LastUpdate = Now;
  }

  else if (_SkinId != "") {
    Persistent_LibCustomSkins_ModelIdToSkinId[Index] = _SkinId;
    Persistent_LibCustomSkins_LastUpdate = Now;
  }
}

/**
 * Set custom skin entry for a model.
 */
Void SetPlayerSkinId(Text _ModelId, Text _SkinId) {
  SetPlayerSkinId(_ModelId, Common::C_DefaultSkinCategoryId, _SkinId);
}

/**
 * Remove custom skin entry for a model in a given category.
 */
Void UnsetPlayerSkinId(Text _ModelId, Text _SkinCategoryId) {
  SetPlayerSkinId(_ModelId, _SkinCategoryId, "");
}

/**
 * Remove custom skin entry for a model.
 */
Void UnsetPlayerSkinId(Text _ModelId) {
  UnsetPlayerSkinId(_ModelId, Common::C_DefaultSkinCategoryId);
}

/**
 * Get the id of the selected skin for a model in a given category.
 */
Text GetPlayerSkinId(Text _ModelId, Text _SkinCategoryId) {
  declare persistent Text[Text] Persistent_LibCustomSkins_ModelIdToSkinId for LocalUser;

  declare Index = Common::GetIndex(_ModelId, _SkinCategoryId);

  if (!Persistent_LibCustomSkins_ModelIdToSkinId.existskey(Index)) return "";

  return Persistent_LibCustomSkins_ModelIdToSkinId[Index];
}

/**
 * Get the id of the selected skin for a model.
 */
Text GetPlayerSkinId(Text _ModelId) {
  return GetPlayerSkinId(_ModelId, Common::C_DefaultSkinCategoryId);
}

/**
 * Set if the modes should spawn in a custom skin instead of an archetype.
 * This setting does not exist for skin categories, because those use custom skins only.
 */
Void SetPlayerUseCustomOverArchetype(Text _ModelId, Boolean _PreferCustom) {
  declare persistent Integer Persistent_LibCustomSkins_LastUpdate for LocalUser;
  declare persistent Text[] Persistent_LibCustomSkins_UseCustomOverArchetype for LocalUser;

  if (_PreferCustom && !Persistent_LibCustomSkins_UseCustomOverArchetype.exists(_ModelId)) {
    Persistent_LibCustomSkins_UseCustomOverArchetype.add(_ModelId);
    Persistent_LibCustomSkins_LastUpdate = Now;
  }

  else if (!_PreferCustom && Persistent_LibCustomSkins_UseCustomOverArchetype.exists(_ModelId)) {
    declare Removed = Persistent_LibCustomSkins_UseCustomOverArchetype.remove(_ModelId);

    Persistent_LibCustomSkins_LastUpdate = Now;
  }
}

/**
 * Check if the modes should spawn in a custom skin instead of an archetype.
 */
Boolean GetPlayerUseCustomOverArchetype(Text _ModelId) {
  declare persistent Text[] Persistent_LibCustomSkins_UseCustomOverArchetype for LocalUser;

  return Persistent_LibCustomSkins_UseCustomOverArchetype.exists(_ModelId);
}
