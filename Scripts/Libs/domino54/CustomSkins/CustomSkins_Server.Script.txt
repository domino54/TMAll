#Const Version      "2024-12-01"
#Const ScriptName   "Libs/domino54/CustomSkins/CustomSkins_Server.Script.txt"

// ---------------------------------- //
// Global variables
// ---------------------------------- //

declare CMode G_Mode;

// ---------------------------------- //
// Private functions
// ---------------------------------- //

CUIConfig Private_GetUI(CUser _User) {
  if (_User == Null) return Null;
  return G_Mode.UIManager.GetUI(_User);
}

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Load(CMode _Mode) {
  G_Mode <=> _Mode;
}

Text GetPlayerSkinId(CUser _User, Text _ModelId) {
  declare UI <=> Private_GetUI(_User);

  if (UI != Null) {
    declare netread Text[Text] Net_LibCustomSkins_ModelIdToSkinId for UI;

    if (Net_LibCustomSkins_ModelIdToSkinId.existskey(_ModelId)) {
      return Net_LibCustomSkins_ModelIdToSkinId[_ModelId];
    }
  }

  // Don't use persistent values on servers.
  if (G_Mode.DataFileMgr == Null) {
    return "";
  }

  declare persistent Text[Text] Persistent_LibCustomSkins_ModelIdToSkinId for _User;

  if (Persistent_LibCustomSkins_ModelIdToSkinId.existskey(_ModelId)) {
    return Persistent_LibCustomSkins_ModelIdToSkinId[_ModelId];
  }

  return "";
}

Text GetPlayerSkinId(CPlayer _Player, Text _ModelId) {
  return GetPlayerSkinId(_Player.User, _ModelId);
}

Boolean GetPlayerUseCustomOverArchetype(CUser _User, Text _ModelId) {
  declare UI <=> Private_GetUI(_User);

  if (UI != Null) {
    declare netread Text[] Net_LibCustomSkins_UseCustomOverArchetype for UI;

    if (Net_LibCustomSkins_UseCustomOverArchetype.exists(_ModelId)) {
      return True;
    }
  }

  // Don't use persistent values on servers.
  if (G_Mode.DataFileMgr == Null) {
    return False;
  }

  declare persistent Text[] Persistent_LibCustomSkins_UseCustomOverArchetype for _User;

  return Persistent_LibCustomSkins_UseCustomOverArchetype.exists(_ModelId);
}

Boolean GetPlayerUseCustomOverArchetype(CPlayer _Player, Text _ModelId) {
  return GetPlayerUseCustomOverArchetype(_Player.User, _ModelId);
}
