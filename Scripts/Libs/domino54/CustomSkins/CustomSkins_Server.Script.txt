#Include "Libs/domino54/CustomSkins/LB_CustomSkins.Script.txt" as LB_CustomSkins

// ---------------------------------- //
// Constants
// ---------------------------------- //

declare CMode G_Mode;
declare Text[Text][Text] G_LibCustomSkins_PlayerModelIdToSkinId;
declare Text[][Text] G_LibCustomSkins_PlayerUseCustomOverArchetype;

// ---------------------------------- //
// Private functions
// ---------------------------------- //

CPlayer Private_GetPlayer(CUIConfig _UI) {
  switchtype (G_Mode) {
    case CTmMode: {
      return (G_Mode as CTmMode).GetPlayer(_UI);
    }
    case CSmMode: {
      return (G_Mode as CTmMode).GetPlayer(_UI);
    }
  }

  return Null;
}

Void Private_ReadPlayerSkinSettings(CUser _User) {
  if (_User == Null) return;

  declare persistent Text[Text] Persistent_LibCustomSkins_ModelIdToSkinId for _User;
  declare persistent Text[] Persistent_LibCustomSkins_UseCustomOverArchetype for _User;

  G_LibCustomSkins_PlayerModelIdToSkinId[_User.Login] = Persistent_LibCustomSkins_ModelIdToSkinId;
  G_LibCustomSkins_PlayerUseCustomOverArchetype[_User.Login] = Persistent_LibCustomSkins_UseCustomOverArchetype;
}

/**
 * Fetches data in solo context.
 */
 Void Private_LoadLocalData() {
  foreach (User in Users) {
    declare persistent Integer Persistent_LibCustomSkins_LastUpdate for User;
    declare Integer LibCustomSkins_PrevUpdate for User;

    if (LibCustomSkins_PrevUpdate == Persistent_LibCustomSkins_LastUpdate) continue;
    LibCustomSkins_PrevUpdate = Persistent_LibCustomSkins_LastUpdate;

    Private_ReadPlayerSkinSettings(User);
  }
}

/**
 * Fetches data in online context.
 */
Void Private_OnlineDataLoop() {
  foreach (Event in G_Mode.UIManager.PendingEvents) {
    if (Event.Type != CUIConfigEvent::EType::OnLayerCustomEvent || Event.CustomEventType != LB_CustomSkins::C_PlayerDataUpdateId || Event.CustomEventData.count < 2) {
      continue;
    }

    declare Player <=> Private_GetPlayer(Event.UI);

    if (Player == Null) {
      continue;
    }

    declare Text[Text] ModelIdToSkinId;
    declare Text[] UseCustomOverArchetype;

    declare Copied0 = ModelIdToSkinId.fromjson(Event.CustomEventData[0]);
    declare Copied1 = UseCustomOverArchetype.fromjson(Event.CustomEventData[1]);

    G_LibCustomSkins_PlayerModelIdToSkinId[Player.User.Login] = ModelIdToSkinId;
    G_LibCustomSkins_PlayerUseCustomOverArchetype[Player.User.Login] = UseCustomOverArchetype;
  }
}

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Unload() {
  foreach (User in Users) {
    declare Integer LibCustomSkins_PrevUpdate for User;
    LibCustomSkins_PrevUpdate = -1;
  }
}

Void Load(CMode _Mode) {
  Unload();
  Private_LoadLocalData();

  G_Mode <=> _Mode;
}

Void LoadLocalData() {
  Private_LoadLocalData();
}

Void OnlineYield() {
  Private_OnlineDataLoop();
}

Text GetPlayerSkinId(Text _Login, Text _ModelId) {
  if (!G_LibCustomSkins_PlayerModelIdToSkinId.existskey(_Login) || !G_LibCustomSkins_PlayerModelIdToSkinId[_Login].existskey(_ModelId)) {
    return "";
  }

  return G_LibCustomSkins_PlayerModelIdToSkinId[_Login][_ModelId];
}

Boolean GetPlayerUseCustomOverArchetype(Text _Login, Text _ModelId) {
  return G_LibCustomSkins_PlayerUseCustomOverArchetype.existskey(_Login) && G_LibCustomSkins_PlayerUseCustomOverArchetype[_Login].exists(_ModelId);
}
