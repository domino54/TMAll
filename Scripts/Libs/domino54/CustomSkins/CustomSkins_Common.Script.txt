#Const C_DefaultSkinCategoryId "default"

// ---------------------------------- //
// Public functions
// ---------------------------------- //

/**
 * Get index for a stored skin entry.
 */
Text GetIndex(Text _ModelId, Text _SkinCategoryId) {
  if (_SkinCategoryId == "" || _SkinCategoryId == C_DefaultSkinCategoryId) {
    return _ModelId;
  }

  return """{{{_ModelId}}}:{{{_SkinCategoryId}}}""";
}
