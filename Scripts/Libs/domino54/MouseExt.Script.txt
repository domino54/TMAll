// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_DefaultDragSensitivity   6.0

// ---------------------------------- //
// Structs
// ---------------------------------- //

/**
 * Options passed to the events function.
 * All text fields are parts of the manialink script executed when a corresponding "event" is fired.
 * Think of it like a glorified switch statement.
 */
#Struct K_Options {
  /**
   * Fired when left mouse button is released on the clicked control.
   * Variables: CMlControl EventControl, Vec2 EventOrigin.
   */
  Text MouseUp;
  /**
   * Fired when right mouse button is pressed on the hovered control.
   * Variables: CMlControl EventControl, Vec2 EventOrigin.
   */
  Text MouseRightDown;
  /**
   * Fired when right mouse button is released on the clicked control.
   * Variables: CMlControl EventControl, Vec2 EventOrigin.
   */
  Text MouseRightUp;
  /**
   * Optional parameter for minimum distance required to start dragging.
   */
  Real DragMin;
  /**
   * Fired when a control starts being dragged.
   * Variables: CMlControl EventControl, Vec2 EventOrigin.
   */
  Text DragStart;
  /**
   * Fired when a control is dragged and the position has changed.
   * Variables: CMlControl EventControl, Vec2 EventOrigin, Vec2 EventVelocity.
   */
  Text DragMove;
  /**
   * Fired when a control stops being dragged.
   * Variables: CMlControl EventControl, Vec2 EventOrigin, Vec2 EventVelocity.
   */
  Text DragEnd;
}

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Text InjectDeclares() {
  return """
declare CMlControl G_MouseClickControl;
declare Vec2 G_MouseClickOrigin;

declare CMlControl G_MouseDragControl;
declare Vec2 G_MouseDragOrigin;
declare Vec2 G_MouseDragPrev;

declare Boolean G_MouseRightPrev;
declare CMlControl G_MouseRightControl;
declare Vec2 G_MouseRightOrigin;
  """;
}

Text InjectEvents(K_Options _Options) {
  declare DragMin = _Options.DragMin;
  declare HasDrag = _Options.DragStart != "" || _Options.DragMove != "" || _Options.DragEnd != "";
  declare HasMouseRight = _Options.MouseRightDown != "" || _Options.MouseRightUp != "";

  if (DragMin == 0.0) {
    DragMin = C_DefaultDragSensitivity;
  }

  declare Result = """
foreach (Event in PendingEvents) {
  if (Event.Type == CMlScriptEvent::Type::MouseClick) {
    G_MouseClickControl = Event.Control;
    G_MouseClickOrigin = Input.MousePos;
  }
}
  """;

  if (HasMouseRight) {
    Result ^= """
if (G_MouseRightPrev != Input.MouseRightButton) {
  G_MouseRightPrev = Input.MouseRightButton;

  if (Input.MouseRightButton && Page.FocusedControl != Null) {
    G_MouseRightControl = Page.FocusedControl;
    G_MouseRightOrigin = Input.MousePos;

    declare EventControl = G_MouseRightControl;
    declare EventOrigin = G_MouseRightOrigin;

    {{{_Options.MouseRightDown}}}
  }

  else if (!Input.MouseRightButton && G_MouseRightControl != Null) {
    if (Page.FocusedControl == G_MouseRightControl) {
      declare EventControl = G_MouseRightControl;
      declare EventOrigin = G_MouseRightOrigin;

      {{{_Options.MouseRightUp}}}
    }

    G_MouseRightControl = Null;
  }
}
    """;
  }

  if (HasDrag) {
    Result ^= """
if (G_MouseClickControl != Null) {
  if (Input.MouseLeftButton && G_MouseDragControl == Null && !G_MouseClickControl.DataAttributeExists("nodrag") && (ML::Abs(Input.MousePos.X - G_MouseClickOrigin.X) > {{{DragMin}}} || ML::Abs(Input.MousePos.Y - G_MouseClickOrigin.Y) > {{{DragMin}}})) {
    G_MouseDragControl = G_MouseClickControl;
    G_MouseDragOrigin = Input.MousePos;
    G_MouseDragPrev = Input.MousePos;

    G_MouseClickControl = Null;

    declare EventControl = G_MouseClickControl;
    declare EventOrigin = G_MouseClickOrigin;

    {{{_Options.DragStart}}}
  }
}
    """;
  }

  Result ^= """
if (G_MouseClickControl != Null && !Input.MouseLeftButton) {
  if (Page.FocusedControl == G_MouseClickControl) {
    declare EventControl = G_MouseClickControl;
    declare EventOrigin = G_MouseClickOrigin;

    {{{_Options.MouseUp}}}
  }

  G_MouseClickControl = Null;
}
  """;

  if (HasDrag) {
    Result ^= """
if (G_MouseDragControl != Null) {
  declare EventControl = G_MouseDragControl;
  declare EventOrigin = G_MouseDragOrigin;
  declare Vec2 EventVelocity;

  if (G_MouseDragPrev != Input.MousePos) {
    EventVelocity = (Input.MousePos - G_MouseDragPrev) * (1000.0 / Period);

    G_MouseDragPrev = Input.MousePos;

    {{{_Options.DragMove}}}
  }

  if (!Input.MouseLeftButton) {
    G_MouseDragControl = Null;

    {{{_Options.DragEnd}}}
  }
}
    """;
  }

  return Result;
}
