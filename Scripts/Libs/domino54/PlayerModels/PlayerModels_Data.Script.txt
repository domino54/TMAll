/**
 * Remove everything after the last slash, forward or back.
 */
Text Private_GetBaseUrl(Text _Url) {
  return TL::RegexReplace(C_RegexFileNameFull, _Url, "i", "");
}

/**
 * Checks if a string begins with ./ or .\ indicating a relative path.
 */
Boolean Private_IsRelativeUrl(Text _Value) {
  return TL::StartsWith("./", _Value) || TL::StartsWith(".\\", _Value);
}

/**
 * Glue a relative path to the base url.
 * Assumes that `_RelativePath` starts with "./" as the first 2 characters are skipped.
 */
Text Private_FromBaseUrl(Text _BaseUrl, Text _RelativePath) {
  return _BaseUrl ^ TL::SubString(_RelativePath, 2, TL::Length(_RelativePath) - 2);
}

/**
 * Removes the namespace from an id.
 *
 * `trackmania:canyoncar` -> `canyoncar`
 * `stadiumcar` -> `stadiumcar`
 * `malformed:name:text` -> `name`
 */
Text Private_TrimNamespace(Text _Id) {
  declare Split = TL::Split(":", _Id);

  if (Split.existskey(1)) {
    return Split[1];
  }

  return _Id;
}

/**
 * Normalizes a file path to a just the file name.
 * This name is used later to look up the custom model id.
 *
 * `StadiumCar` -> `StadiumCar`
 * `CanyonCar.Item.Gbx` -> `CanyonCar`
 * `Vehicles/IslandCar.Item.Gbx` -> `IslandCar`
 */
Text Private_GetItemNameWithoutExtension(Text _ItemPath) {
  declare Match = TL::RegexMatch(C_RegexItemName, _ItemPath, "i");

  if (Match.count < 2 || Match[1] == "") {
    return _ItemPath;
  }

  return Match[1];
}

/**
 * Normalizes the string to use forward slashes.
 * @example `Vehicles\\CanyonCar.Item.Gbx` -> `Vehicles/CanyonCar.Item.Gbx`
 */
Text Private_NormalizeSlashes(Text _ItemPath) {
  return TL::ToLowerCase(TL::RegexReplace(C_RegexReplacedSlashes, _ItemPath, "i", "/"));
}

/**
 * Check if an archetype is a character.
 */
Boolean Private_IsCharacter(Text _Archetype) {
  return C_CharacterArchetypes.exists(_Archetype);
}

/**
 * Check if a skin applies to a base model.
 */
Boolean Private_SkinAppliesTo(Text _AppliesTo, Text _ModelId) {
  return _AppliesTo == _ModelId || _AppliesTo == C_CarCommonWildcard;
}

/**
 * Check if a skin can be used within a category.
 * Empty array is treated as if it contains the default category.
 */
Boolean Private_IsSkinForCategory(Text[] _SkinCategoryIds, Text _TargetCategoryId) {
  if (_SkinCategoryIds.count <= 0 && _TargetCategoryId == C_DefaultSkinCategoryId) {
    return True;
  }

  return _SkinCategoryIds.exists(_TargetCategoryId);
}

/**
 * Pretty much a spread operator in a `{ ...target, ...source }` fashion.
 */
Text[Text] Private_AppendAssets(Text[Text] _Target, Text[Text] _Source, Text _BaseUrl) {
  declare Result = _Target;

  foreach (Key => Value in _Source) {
    if (Private_IsRelativeUrl(Value)) {
      Result[Key] = Private_FromBaseUrl(_BaseUrl, Value);
    } else {
      Result[Key] = Value;
    }
  }

  return Result;
}

/**
 * Register a new model, or replace existing fields if a value is given.
 */
Void Private_AppendModel(K_Model _Model, Text _BaseUrl) {
  if (G_LibPlayerModels_ModelById.existskey(_Model.Id)) {
    if (_Model.Name != "") {
      G_LibPlayerModels_ModelById[_Model.Id].Name = _Model.Name;
    }

    if (_Model.Archetype != "") {
      G_LibPlayerModels_ModelById[_Model.Id].Archetype = _Model.Archetype;
    }

    if (_Model.ItemPath != "") {
      G_LibPlayerModels_ModelById[_Model.Id].ItemPath = _Model.ItemPath;
    }

    if (_Model.LoadAs != "") {
      G_LibPlayerModels_ModelById[_Model.Id].LoadAs = _Model.LoadAs;
    }

    if (_Model.Author != "") {
      G_LibPlayerModels_ModelById[_Model.Id].Author = _Model.Author;
    }
  }
  else {
    G_LibPlayerModels_ModelById[_Model.Id] = _Model;
  }

  if (_Model.Assets.count > 0) {
    G_LibPlayerModels_ModelById[_Model.Id].Assets = Private_AppendAssets(G_LibPlayerModels_ModelById[_Model.Id].Assets, _Model.Assets, _BaseUrl);
  }
}

/**
 * Register a new skin, or replace existing fields if a value is given.
 */
Void Private_AppendSkin(K_Skin _Skin, Text _BaseUrl) {
  if (G_LibPlayerModels_SkinById.existskey(_Skin.Id)) {
    if (_Skin.Name != "") {
      G_LibPlayerModels_SkinById[_Skin.Id].Name = _Skin.Name;
    }

    if (_Skin.AppliesTo != "") {
      G_LibPlayerModels_SkinById[_Skin.Id].AppliesTo = _Skin.AppliesTo;
    }

    if (_Skin.SkinPath != "") {
      G_LibPlayerModels_SkinById[_Skin.Id].SkinPath = _Skin.SkinPath;
    }

    if (_Skin.Author != "") {
      G_LibPlayerModels_SkinById[_Skin.Id].Author = _Skin.Author;
    }

    if (_Skin.Rarity > 0) {
      G_LibPlayerModels_SkinById[_Skin.Id].Rarity = _Skin.Rarity;
    }

    if (_Skin.SortIndex != 0) {
      G_LibPlayerModels_SkinById[_Skin.Id].Rarity = _Skin.SortIndex;
    }
  }
  else {
    G_LibPlayerModels_SkinById[_Skin.Id] = _Skin;
  }

  if (_Skin.Assets.count > 0) {
    G_LibPlayerModels_SkinById[_Skin.Id].Assets = Private_AppendAssets(G_LibPlayerModels_SkinById[_Skin.Id].Assets, _Skin.Assets, _BaseUrl);
  }
}

/**
 * Get the model from a skin directory.
 */
Text Private_ResolvePlayerModelIdFromSkinDirectory(Text _SkinDirectory) {
  // CarCommon quick check (direct).
  if (_SkinDirectory == C_CarCommonDirectory) {
    return C_CarCommonWildcard;
  }

  // The skin file sits directly in a defined folder.
  if (G_LibPlayerModels_ModelIdBySkinDirectory.existskey(_SkinDirectory)) {
    return G_LibPlayerModels_ModelIdBySkinDirectory[_SkinDirectory];
  }

  // For subdirectories, match the most specific path.
  foreach (SkinDirectory => ModelId in G_LibPlayerModels_ModelIdBySkinDirectory) {
    if (TL::StartsWith(_SkinDirectory, SkinDirectory, True, False)) {
      return ModelId;
    }
  }

  // CarCommon slow check (subfolder).
  if (TL::StartsWith(_SkinDirectory, C_CarCommonDirectory, True, False)) {
    return C_CarCommonWildcard;
  }

  return "";
}

/**
 * Parse user directory skin path and append a skin if a matching model is found.
 */
K_Skin Private_CreateSkinFromPath(Text _SkinPath) {
  declare Match = TL::RegexMatch(C_RegexMatchUserSkinPath, _SkinPath, "i");

  if (Match.count != 3) {
    return K_Skin{};
  }

  declare ModelId = Private_ResolvePlayerModelIdFromSkinDirectory(Match[1]^"\\");

  if (ModelId == "") {
    return K_Skin{};
  }

  return K_Skin{
    IsCustom = True,
    Id = """{{{C_UserSkinsNamespace}}}:{{{_SkinPath}}}""",
    AppliesTo = ModelId,
    Name = Match[2],
    SkinPath = """file://Skins/{{{TL::Replace(_SkinPath, "\\", "/")}}}"""
  };
}

/**
 * Add a lowercase alias to a model id.
 */
Void Private_AppendModelIdAlias(Text _ModelId, Text _Alias) {
  if (_ModelId == "" || _Alias == "") {
    return;
  }

  declare Alias = TL::ToLowerCase(_Alias);

  if (G_LibPlayerModels_ModelIdByAlias.existskey(Alias)) {
    return;
  }

  G_LibPlayerModels_ModelIdByAlias[Alias] = _ModelId;
}

/**
 * Bulk create aliases for a model.
 */
Void Private_CreateModelAliases(K_Model[Text] _ModelById) {
  foreach (ModelId => Model in _ModelById) {
    Private_AppendModelIdAlias(ModelId, ModelId); // Case-insensitive alias.
    Private_AppendModelIdAlias(ModelId, Private_TrimNamespace(ModelId));
    Private_AppendModelIdAlias(ModelId, Private_GetItemNameWithoutExtension(Model.Archetype));
    Private_AppendModelIdAlias(ModelId, Private_NormalizeSlashes(Model.ItemPath));
    Private_AppendModelIdAlias(ModelId, Private_GetItemNameWithoutExtension(Model.ItemPath));
  }
}

/**
 * Bulk create aliases for an overlay.
 */
Void Private_CreateOverlayAliases(K_Overlay[Text] _OverlayById) {
  foreach (OverlayId => Overlay in _OverlayById) {
    Private_AppendModelIdAlias(OverlayId, OverlayId); // Case-insensitive alias.
    Private_AppendModelIdAlias(OverlayId, Private_TrimNamespace(OverlayId));
    Private_AppendModelIdAlias(OverlayId, Private_NormalizeSlashes(Overlay.ItemPath));
    Private_AppendModelIdAlias(OverlayId, Private_GetItemNameWithoutExtension(Overlay.ItemPath));
  }
}

/**
 * Once every source file has been loaded, load the data from all of them at once.
 */
Void Private_ProcessSourceFiles(K_SourceFile[] _SourceFiles, Text[] _UserDirectorySkins) {
  // Used exclusively for the display naming convention.
  declare NbCharacters = 0;
  declare NbVehicles = 0;

  // Source files are iterated two separate times, and that is intended.
  // We want to be absolutely sure that all base models are loaded before everything else.
  foreach (SourceFile in _SourceFiles) {
    foreach (Model in SourceFile.Models) {
      if (Model.Id == "") {
        continue;
      }

      foreach (SkinDirectory in Model.SkinDirectories) {
        G_LibPlayerModels_ModelIdBySkinDirectory[SkinDirectory^"\\"] = Model.Id;
      }

      Private_AppendModel(Model, SourceFile.BaseUrl);

      if (Private_IsCharacter(Model.Archetype)) {
        NbCharacters += 1;
      } else {
        NbVehicles += 1;
      }
    }
  }

  // Reverse sorting allows the deepest subdirectories to be checked first.
  G_LibPlayerModels_ModelIdBySkinDirectory = G_LibPlayerModels_ModelIdBySkinDirectory.sortkeyreverse();

  // Append skins and overlays.
  foreach (SourceFile in _SourceFiles) {
    foreach (Skin in SourceFile.Skins) {
      if (Skin.Id == "" || (!G_LibPlayerModels_ModelById.existskey(Skin.AppliesTo) && Skin.Id != C_CarCommonWildcard)) {
        continue;
      }

      Private_AppendSkin(Skin, SourceFile.BaseUrl);
    }

    foreach (SkinCategory in SourceFile.SkinCategories) {
      Private_AppendSkinCategory(SkinCategory, SourceFile.BaseUrl);
    }

    foreach (Overlay in SourceFile.Overlays) {
      if (Overlay.Id == "" || !G_LibPlayerModels_ModelById.existskey(Overlay.Extends)) {
        continue;
      }

      Private_AppendOverlay(Overlay, SourceFile.BaseUrl);
    }
  }

  // Creates dummy skin entries from the user Skins directory files.
  foreach (File in _UserDirectorySkins) {
    declare Skin = Private_CreateSkinFromPath(File);

    if (Skin.Id == "") {
      continue;
    }

    Private_AppendSkin(Skin, "");
  }

  // Create aliases.
  Private_CreateModelAliases(G_LibPlayerModels_ModelById);
  Private_CreateOverlayAliases(G_LibPlayerModels_OverlayById);

  // Get the metrics.
  declare NbModels = G_LibPlayerModels_ModelById.count;
  declare NbSkins = G_LibPlayerModels_SkinById.count;
  declare NbSkinCategories = G_LibPlayerModels_SkinCategoryById.count;
  declare NbOverlays = G_LibPlayerModels_OverlayById.count;

  // Apply the correct naming convention, depending on the loaded model types.
  if (NbModels == NbCharacters) {
    G_LibPlayerModels_ModelTypeCategory = C_ModelType_Character;
  }

  if (NbModels == NbVehicles) {
    G_LibPlayerModels_ModelTypeCategory = C_ModelType_Vehicle;
  }

  Private_Log("""Loaded data of {{{NbModels}}} models, {{{NbSkins}}} skins, {{{NbSkinCategories}}} skin categories, {{{NbOverlays}}} overlays""");
}

/**
 * Clear all the cached data.
 */
Void Private_ClearData() {
  declare DefaultSkinCategory = K_SkinCategory{
    Id = C_DefaultSkinCategoryId,
    Name = _("Default")
  };

  G_LibPlayerModels_SortingOverrides.clear();
  G_LibPlayerModels_OverlayById.clear();
  G_LibPlayerModels_ModelById.clear();
  G_LibPlayerModels_SkinById.clear();
  G_LibPlayerModels_SkinCategoryById = [DefaultSkinCategory.Id => DefaultSkinCategory];
  G_LibPlayerModels_ModelIdByAlias.clear();
  G_LibPlayerModels_ModelIdBySkinDirectory.clear();
}

/**
 * Synchronously fetch a file by URL, blocking the main thread.
 */
Text Private_FetchFile(Text _Url) {
  declare UseCache = Env::Get() == Env::C_Env_Production;
  declare Request = Http.CreateGet(_Url, UseCache);

  // In 99% cases this will not wait any script frame, but will load synchronously instead.
  wait(Request.IsCompleted);

  declare Result = Request.Result;

  Http.Destroy(Request);

  return Result;
}

/**
 * Fetch a file and parse it as an index file.
 */
K_IndexFile Private_FetchIndexFile(Text _Url) {
  declare K_IndexFile IndexFile;
  declare Copied = IndexFile.fromjson(Private_FetchFile(_Url)); // @mslint-disable-line no-unused-variables

  return IndexFile;
}

/**
 * Fetch a file and parse it as a source file.
 */
K_SourceFile Private_FetchSourceFile(Text _Url) {
  declare K_SourceFile SourceFile;
  declare Copied = SourceFile.fromjson(Private_FetchFile(_Url)); // @mslint-disable-line no-unused-variables

  SourceFile.BaseUrl = Private_GetBaseUrl(_Url);

  return SourceFile;
}

/**
 * Load the player models data synchronously with the given options.
 */
Void Private_Load(K_Options _Options) {
  declare IndexUrl = C_DefaultIndexFileUrl;

  if (_Options.IndexFileUrl != "") {
    IndexUrl = _Options.IndexFileUrl;
  }

  if (!Http.IsValidUrl(IndexUrl)) {
    return;
  }

  // Wipe existing data.
  Private_ClearData();

  declare IndexFile = Private_FetchIndexFile(IndexUrl);

  // These are added on top of everything, if the game mode wants to load extra models.
  foreach (ExtraSourceUrl in _Options.ExtraSourceFilesUrls) {
    IndexFile.SourceUrls.add(ExtraSourceUrl);
  }

  if (IndexFile.SourceUrls.count <= 0) {
    return;
  }

  G_LibPlayerModels_SortingOverrides = IndexFile.SortOverrides;

  declare BaseUrl = Private_GetBaseUrl(IndexUrl);

  // Fetch all source files.
  declare K_SourceFile[] SourceFilesToProcess;

  foreach (RawSourceUrl in IndexFile.SourceUrls) {
    declare SourceUrl = RawSourceUrl;

    if (Private_IsRelativeUrl(SourceUrl)) {
      SourceUrl = Private_FromBaseUrl(BaseUrl, SourceUrl);
    }

    if (!Http.IsValidUrl(SourceUrl)) {
      continue;
    }

    SourceFilesToProcess.add(Private_FetchSourceFile(SourceUrl));
  }

  // Custom skins (for selection interface only).
  declare Text[] UserDirectorySkins;

  if (_Options.LoadUserSkinsDirectory && DataFileMgr != Null) {
    foreach (Directory in C_UserSkinsDirectories) {
      declare Task <=> Private_CreateUserDataTask(Directory);

      // Yes, this is bad.
      // This only happens in the Garage UI and Garage mode.
      wait(!Task.IsProcessing);

      foreach (File in Task.Files) {
        UserDirectorySkins.add(File);
      }

      DataFileMgr.TaskResult_Release(Task.Id);
    }
  }

  Private_ProcessSourceFiles(SourceFilesToProcess, UserDirectorySkins);
}

/**
 * @returns Custom model id resolved from the an item path, item name or model archetype.
 * @example `Vehicles/IslandCar.Item.Gbx` -> `trackmania:islandcar`
 */
Text ResolveModelId(Text _ItemPath) {
  if (_ItemPath == "") {
    return _ItemPath;
  }

  // Always run the check case-insensitive.
  declare ItemPath = TL::ToLowerCase(_ItemPath);

  if (G_LibPlayerModels_ModelIdByAlias.existskey(ItemPath)) {
    return G_LibPlayerModels_ModelIdByAlias[ItemPath];
  }

  // Desperately try to match some loaded item.
  declare NormalizedItemPath = Private_NormalizeSlashes(ItemPath);

  return G_LibPlayerModels_ModelIdByAlias.get(NormalizedItemPath, "");
}

/**
 * @returns Existing skin id or skin category id of the given value.
 */
K_SkinIdResult ResolveSkinId(Text _Value) {
  if (_Value == "") {
    return K_SkinIdResult{};
  }

  // Category direct hit!
  if (G_LibPlayerModels_SkinCategoryById.existskey(_Value)) {
    return K_SkinIdResult{ SkinCategoryId = _Value };
  }

  // Skin direct hit!
  if (G_LibPlayerModels_SkinById.existskey(_Value)) {
    return K_SkinIdResult{ SkinId = _Value };
  }

  return K_SkinIdResult{};
}

/**
 * @returns Base custom model id of the gameplay extension, or the custom model id itself.
 */
Text GetBaseModelId(Text _ModelId) {
  if (G_LibPlayerModels_OverlayById.existskey(_ModelId)) {
    return G_LibPlayerModels_OverlayById[_ModelId].Extends;
  }

  if (G_LibPlayerModels_ModelById.existskey(_ModelId)) {
    return _ModelId;
  }

  return "";
}

K_Model[] GetModelsList(Boolean _IncludeHidden) {
  declare Text[] SortedModelIds;

  foreach (ModelId in G_LibPlayerModels_SortingOverrides) {
    if (G_LibPlayerModels_ModelById.existskey(ModelId)) {
      SortedModelIds.add(ModelId);
    }
  }

  foreach (ModelId => __ in G_LibPlayerModels_ModelById) {
    if (!SortedModelIds.exists(ModelId)) {
      SortedModelIds.add(ModelId);
    }
  }

  declare K_Model[] Result;

  foreach (ModelId in SortedModelIds) {
    if (!G_LibPlayerModels_ModelById.existskey(ModelId)) {
      continue;
    }

    declare Model = G_LibPlayerModels_ModelById[ModelId];

    if (!_IncludeHidden && Model.IsHidden) {
      continue;
    }

    Result.add(Model);
  }

  return Result;
}

K_Model[] GetModelsList() {
  return GetModelsList(False);
}

K_Skin[] GetSkinsListByModelId(Text _ModelId, Text _SkinCategoryId, Boolean _IncludeHidden) {
  declare BaseModelId = GetBaseModelId(_ModelId);
  declare K_Skin[] Result;

  foreach (Skin in G_LibPlayerModels_SkinById) {
    if (!Private_SkinAppliesTo(Skin.AppliesTo, BaseModelId)) {
      continue;
    }

    if (_SkinCategoryId != "" && !Private_IsSkinForCategory(Skin.Categories, _SkinCategoryId)) {
      continue;
    }

    if (!_IncludeHidden && Skin.IsHidden) {
      continue;
    }

    Result.add(Skin);
  }

  return Result;
}

K_Skin[] GetSkinsListByModelId(Text _ModelId, Text _SkinCategoryId) {
  return GetSkinsListByModelId(_ModelId, _SkinCategoryId, False);
}

K_Skin[] GetSkinsListByModelId(Text _ModelId, Boolean _IncludeHidden) {
  return GetSkinsListByModelId(_ModelId, "", _IncludeHidden);
}

K_Skin[] GetSkinsListByModelId(Text _ModelId) {
  return GetSkinsListByModelId(_ModelId, False);
}

Text GetRandomSkinIdByModelId(Text _ModelId, Text _SkinCategoryId, Boolean _IncludeHidden) {
  declare Skins = GetSkinsListByModelId(_ModelId, _SkinCategoryId, _IncludeHidden);

  if (Skins.count <= 0) {
    return "";
  }

  if (Skins.count == 1) {
    return Skins[0].Id;
  }

  declare Index = ML::Rand(0, Skins.count - 1);

  return Skins[Index].Id;
}

Text GetModelArchetype(Text _ModelId) {
  declare BaseModelId = GetBaseModelId(_ModelId);

  if (!G_LibPlayerModels_ModelById.existskey(BaseModelId)) {
    return "";
  }

  return G_LibPlayerModels_ModelById[BaseModelId].Archetype;
}

Boolean GetArchetypeIsUnitedVehicle(Text _Archetype) {
  return C_TMUnitedArchetypes.exists(_Archetype);
}

Boolean GetPlayerModelIsUnitedVehicle(Text _ModelId) {
  return GetArchetypeIsUnitedVehicle(GetModelArchetype(_ModelId));
}

K_GameplayInfo GetPlayerModelGameplayInfo(Text _ModelId) {
  declare DefaultResult = K_GameplayInfo{
    AccelCoef = 1.0,
    ControlCoef = 1.0,
    GravityCoef = 1.0
  };

  if (!G_LibPlayerModels_OverlayById.existskey(_ModelId)) {
    return DefaultResult;
  }

  declare GameplayInfo = G_LibPlayerModels_OverlayById[_ModelId].Gameplay;

  if (GameplayInfo.AccelCoef != 0.0 || GameplayInfo.ControlCoef != 0.0 || GameplayInfo.GravityCoef != 0.0) {
    return GameplayInfo;
  }

  return DefaultResult;
}

Text GetModelAsset(Text _ModelId, Text _AssetId, Text _Fallback) {
  if (G_LibPlayerModels_ModelById.existskey(_ModelId)) {
    if (G_LibPlayerModels_ModelById[_ModelId].Assets.existskey(_AssetId)) {
      return G_LibPlayerModels_ModelById[_ModelId].Assets[_AssetId];
    }

    return _Fallback;
  }

  // Fallback to overlay.
  if (!G_LibPlayerModels_OverlayById.existskey(_ModelId)) {
    return _Fallback;
  }

  if (G_LibPlayerModels_OverlayById[_ModelId].Assets.existskey(_AssetId)) {
    return G_LibPlayerModels_OverlayById[_ModelId].Assets[_AssetId];
  }

  // Fallback to overlay base model.
  return GetModelAsset(G_LibPlayerModels_OverlayById[_ModelId].Extends, _AssetId, _Fallback);
}

Text GetSkinAsset(Text _SkinId, Text _AssetId, Text _Fallback) {
  if (!G_LibPlayerModels_SkinById.existskey(_SkinId)) {
    return _Fallback;
  }

  if (G_LibPlayerModels_SkinById[_SkinId].Assets.existskey(_AssetId)) {
    return G_LibPlayerModels_SkinById[_SkinId].Assets[_AssetId];
  }

  // Fallback to model.
  return GetModelAsset(G_LibPlayerModels_SkinById[_SkinId].AppliesTo, _AssetId, _Fallback);
}

K_ModelInfo[] GetAllModelInfo() {
  declare K_ModelInfo[] Result;

  foreach (Model in G_LibPlayerModels_ModelById) {
    Result.add(K_ModelInfo{
      Id = Model.Id,
      BaseId = Model.Id,
      Name = Model.Name,
      Archetype = Model.Archetype,
      ItemPath = Model.ItemPath
    });
  }

  foreach (Overlay in G_LibPlayerModels_OverlayById) {
    declare Model = G_LibPlayerModels_ModelById[Overlay.Extends];

    Result.add(K_ModelInfo{
      Id = Overlay.Id,
      BaseId = Overlay.Extends,
      Name = Overlay.Name,
      Archetype = Model.Archetype,
      ItemPath = Overlay.ItemPath
    });
  }

  return Result;
}
