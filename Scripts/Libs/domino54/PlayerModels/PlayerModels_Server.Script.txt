#Const C_LoadingMode_Compatibility    0   ///< Never loads archetypes or custom items. Replaces all paths with "DefaultPlayerModel".
#Const C_LoadingMode_LoadEverything   1   ///< Loads all archetypes and custom items normally.
#Const C_LoadingMode_AllowArchetypes  2   ///< Same as Compatibility, but allows archetypes to load.

/// These environments restrict the assignable archetypes and require unlock via items.
#Const C_EnvironmentRequiresUnlock ["Stadium"]

/// These archetypes are available without the need of an unlock.
#Const C_VehicleIgnoresUnlock ["StadiumCar"]

/// Model name, which references whatever model is used on the loaded map.
#Const C_DefaultPlayerModel "DefaultPlayerModel"

// Special skin names from TMT, which do not work inside ManiaPlanet and will be ignored.
#Const C_SpecialSkinNames ["", "Default", "Profile"]

// ---------------------------------- //
// Structs
// ---------------------------------- //

#Struct K_Options {
  /**
   * As soon as the game mode loads any items, they will be referenced in the saved replays.
   * This will cause a TON of item references, and make the replays unplayable outside of the title pack.
   * Custom items should load only in game modes, where players can spawn as different models on the same map.
   *
   * Therefore, the compatibility mode will not load any items or archetypes from the actual paths of the source files.
   *
   * Instead, all item paths are replaced with "DefaultPlayerModel", which references the current map model.
   * This model dynamically changes with the current map, but also locks the mode into a single model only.
   */
  Integer LoadingMode;

  /** Ignore model overlay in places, where they are irrelevant. */
  Boolean SkipOverlays;

  /** Load CarCommon skins to the first player model only. Use for skin selection interface. */
  Boolean CarCommonFirstModel;

  /** Load only skins, which the local user has selected (or been assigned). */
  Boolean LocalUserSkinsOnly;

  /** Player model ID allowlist for specific models to be loaded exclusively. */
  Text[] ModelIdAllowlist;
}

#Struct K_ModelIdentResult {
  Ident ModelIdent;
  Boolean IsArchetype;
}

#Struct Structs::K_SkinIdResult as K_SkinIdResult

/**
 * Model skin ID allowlist for specific skins to be loaded exclusively.
 * [SkinId => [ModelId]]

declare Text[][Text] G_SkinIdAllowlist;

/// Lookup table for text-based model ID to loaded item Ident.
declare Ident[Text] G_LoadedItemIdByLookupId;

Text Private_GetLookupId(Text _ModelId, Text _SkinId) {
  return """{{{_ModelId}}}:{{{_SkinId}}}""";
}

Text Private_GetLookupId(Text _ModelId, Boolean _IsArchetype) {
  if (_IsArchetype) {
    return """{{{_ModelId}}}:archetype""";
  }

  return _ModelId;
}

Text Private_GetLookupId(Text _ModelId) {
  return _ModelId;
}

Text Private_GetArchetypePath(Text _ItemPath) {
  if (G_Options.LoadingMode == C_LoadingMode_Compatibility) {
    return C_DefaultPlayerModel;
  }

  return _ItemPath;
}

Text Private_GetItemPath(Text _ItemPath) {
  if (G_Options.LoadingMode != C_LoadingMode_LoadEverything) {
    return C_DefaultPlayerModel;
  }

  return _ItemPath;
}

Text Private_GetRandomEntryInternalIndex(Text _ModelId, Text _SkinCategoryId) {
  if (_SkinCategoryId == "" || _SkinCategoryId == PMData::C_DefaultSkinCategoryId) {
    return _ModelId;
  }

  return """{{{_ModelId}}}:{{{_SkinCategoryId}}}""";
}

/**
 * Pick a random skin for the player and remember it until the title is unloaded.
 */
Text Private_GetPlayerRandomizedSkinId(Text _Login, Text _ModelId, Text _SkinCategoryId) {
  declare Text[Text][Text] LibPlayerModels_RandomizedSkins for LoadedTitle;

  // Login not initialized in the store yet.
  if (!LibPlayerModels_RandomizedSkins.existskey(_Login)) {
    LibPlayerModels_RandomizedSkins[_Login] = [];
  }

  // There's already a random skin assigned to the model id.
  declare Index = Private_GetRandomEntryInternalIndex(_ModelId, _SkinCategoryId);

  if (LibPlayerModels_RandomizedSkins[_Login].existskey(Index)) {
    return LibPlayerModels_RandomizedSkins[_Login][Index];
  }

  // Pick a random skin and keep it on the server side.
  declare SkinId = PMData::GetRandomSkinIdByModelId(_ModelId, _SkinCategoryId);

  if (SkinId != "") {
    LibPlayerModels_RandomizedSkins[_Login][Index] = SkinId;
  }

  return SkinId;
}

/**
 * Get the player skin preference or a randomized skin.
 */
Text Private_GetPlayerSkinId(CUser _User, Text _ModelId, Text _SkinCategoryId) {
  declare BaseModelId = PMData::GetBaseModelId(_ModelId);
  declare PlayerPreferenceId = CustomSkins::GetPlayerSkinId(_User, BaseModelId, _SkinCategoryId);

  // Return the preference if set.
  if (PlayerPreferenceId != "") {
    return PlayerPreferenceId;
  }

  return Private_GetPlayerRandomizedSkinId(_User.Login, BaseModelId, _SkinCategoryId);
}

/**
 * @returns List of all skin IDs the local player has got assigned.
 */
Text[][Text] Private_GetLocalPlayerSkinIds() {
  declare Text[][Text] Result;

  // Users array exists in both game modes and maptypes.
  foreach (User in Users) {
    foreach (BaseModelId => __ in PMData::GetAllModels()) {
      declare PlayerPreferenceId = Private_GetPlayerSkinId(User, BaseModelId, "");

      if (PlayerPreferenceId == "") {
        continue;
      }

      if (Result.existskey(PlayerPreferenceId)) {
        Result[PlayerPreferenceId].add(BaseModelId);
      } else {
        Result[PlayerPreferenceId] = [BaseModelId];
      }
    }
  }

  return Result;
}

Boolean Private_IsModelAllowlisted(Text _ModelId) {
  return G_Options.ModelIdAllowlist.count <= 0 || G_Options.ModelIdAllowlist.exists(_ModelId);
}

Boolean Private_IsSkinAllowlisted(Text _SkinId, Text _ModelId) {
  if (G_SkinIdAllowlist.count <= 0) {
    return True;
  }

  if (_ModelId == "") {
    return G_SkinIdAllowlist.existskey(_SkinId);
  }

  return G_SkinIdAllowlist[_SkinId].exists(_ModelId);
}

Boolean Private_IsSkinAllowlisted(Text _SkinId) {
  return Private_IsSkinAllowlisted(_SkinId, "");
}

Ident Private_ItemListAdd(Text _ItemPath) {
  // No need to load the item at all if it's just default player model.
  if (_ItemPath == C_DefaultPlayerModel) {
    return NullId;
  }

  return G_Mode.ItemList_Add(_ItemPath);
}

Ident Private_ItemListAdd(Text _ItemPath, Text _SkinPath) {
  // Do not load special skin names, as they do not work.
  if (C_SpecialSkinNames.exists(_SkinPath)) {
    return Private_ItemListAdd(_ItemPath);
  }

  return G_Mode.ItemList_AddWithSkin(_ItemPath, _SkinPath);
}

Ident Private_ItemListAdd(Text[] _ItemProps) {
  declare ResultId = NullId;

  switch (_ItemProps.count) {
    case 1: ResultId = Private_ItemListAdd(_ItemProps[0]);
    case 2: ResultId = Private_ItemListAdd(_ItemProps[0], _ItemProps[1]);
    default: assert(False, "Player model could not be loaded.");
  }

  return ResultId;
}

Integer Private_LoadItems(Text[][Text] _ItemsToLoad) {
  declare Ident[Text] CachedItemIds;

  foreach (ModelId => ItemProps in _ItemsToLoad) {
    declare ReusableIndex = TL::Join("*", ItemProps);

    // If we get the same [Model+Skin], don't load the same item twice, reuse an existing Ident.
    if (CachedItemIds.existskey(ReusableIndex)) {
      G_LoadedItemIdByLookupId[ModelId] = CachedItemIds[ReusableIndex];
      continue;
    }

    declare ResultId = Private_ItemListAdd(ItemProps);

    CachedItemIds[ReusableIndex] = ResultId;
    G_LoadedItemIdByLookupId[ModelId] = ResultId;
  }

  return CachedItemIds.count;
}

Void Private_LoadModelItems() {
  // Dictionary of tuples. This is used later to call ItemList_Add.
  // [UniqueId => [ItemPath, SkinPath]]
  declare Text[][Text] ItemsToLoad = [];

  // Holds the item path for loading skins.
  declare Text[Text] BaseModelItemPath = [];

  // Prepare base player models.
  foreach (ModelId => Model in PMData::GetAllModels()) {
    if (!Private_IsModelAllowlisted(ModelId)) {
      continue;
    }

    declare AsItemId = Private_GetLookupId(ModelId);

    if (Model.LoadAs == PMData::C_LoadAs_Archetype) {
      BaseModelItemPath[AsItemId] = Model.Archetype;
    } else {
      BaseModelItemPath[AsItemId] = Model.ItemPath;
    }

    if (Model.LoadAs == PMData::C_LoadAs_ItemAndArchetype || Model.LoadAs == PMData::C_LoadAs_Item) {
      ItemsToLoad[AsItemId] = [Private_GetItemPath(Model.ItemPath)];
    }

    if (Model.LoadAs == PMData::C_LoadAs_ItemAndArchetype || Model.LoadAs == PMData::C_LoadAs_Archetype) {
      declare AsArchetypeId = Private_GetLookupId(ModelId, True);

      ItemsToLoad[AsArchetypeId] = [Private_GetArchetypePath(Model.Archetype)];
    }
  }

  // Limit the loaded skins to the local user preferences.
  if (G_Options.LocalUserSkinsOnly) {
    G_SkinIdAllowlist = Private_GetLocalPlayerSkinIds();
  }

  // Holds skin paths for loading overlays.
  declare Text[Text][Text] SkinsByBaseModelId = [];

  // Prepare skins.
  foreach (SkinId => Skin in PMData::GetAllSkins()) {
    if (!Private_IsSkinAllowlisted(SkinId)) {
      continue;
    }

    if (SkinsByBaseModelId.existskey(Skin.AppliesTo)) {
      SkinsByBaseModelId[Skin.AppliesTo][SkinId] = Skin.SkinPath;
    } else {
      SkinsByBaseModelId[Skin.AppliesTo] = [SkinId => Skin.SkinPath];
    }
  }

  // Regular skins: Load for a single base player model.
  foreach (BaseModelId => Skins in SkinsByBaseModelId) {
    // Must check it here, because overlays deriving from BaseModelId may be allowed.
    if (!Private_IsModelAllowlisted(BaseModelId) || BaseModelId == PMData::C_CarCommonWildcard) {
      continue;
    }

    foreach (SkinId => SkinPath in Skins) {
      declare LookupId = Private_GetLookupId(BaseModelId, SkinId);

      ItemsToLoad[LookupId] = [Private_GetItemPath(BaseModelItemPath[BaseModelId]), SkinPath];
    }
  }

  // CarCommon wildcard: Load the skin for every base player model available.
  declare CarCommonSkins = SkinsByBaseModelId.get(PMData::C_CarCommonWildcard, []);

  foreach (SkinId => SkinPath in CarCommonSkins) {
    foreach (ModelId => __ in BaseModelItemPath) {
      if (!Private_IsSkinAllowlisted(SkinId, ModelId)) {
        continue;
      }

      declare LookupId = Private_GetLookupId(ModelId, SkinId);

      ItemsToLoad[LookupId] = [Private_GetItemPath(BaseModelItemPath[ModelId]), SkinPath];
    }
  }

  // Prepare overlays.
  foreach (ModelOverlayId => ModelOverlay in PMData::GetAllOverlays()) {
    if (G_Options.SkipOverlays) {
      break;
    }

    if (!Private_IsModelAllowlisted(ModelOverlayId)) {
      continue;
    }

    declare BaseModelLookupId = Private_GetLookupId(ModelOverlayId);

    ItemsToLoad[BaseModelLookupId] = [Private_GetItemPath(ModelOverlay.ItemPath)];

    // Overlay is using the same skins as the overlay's base player model.
    declare BaseModelSkins = SkinsByBaseModelId.get(ModelOverlay.Extends, []);

    foreach (SkinId => SkinPath in BaseModelSkins) {
      declare LookupId = Private_GetLookupId(ModelOverlayId, SkinId);

      ItemsToLoad[LookupId] = [Private_GetItemPath(ModelOverlay.ItemPath), SkinPath];
    }

    // If there are any CarCommon skins, apply them to the overlay.
    declare CarCommonSkins = SkinsByBaseModelId.get(PMData::C_CarCommonWildcard, []);

    foreach (SkinId => SkinPath in CarCommonSkins) {
      if (!Private_IsSkinAllowlisted(SkinId, ModelOverlay.Extends)) {
        continue;
      }

      declare LookupId = Private_GetLookupId(ModelOverlayId, SkinId);

      ItemsToLoad[LookupId] = [Private_GetItemPath(ModelOverlay.ItemPath), SkinPath];
    }
  }

  // Load items into the game mode.
  declare NbItems = Private_LoadItems(ItemsToLoad);

  Log::Log("""[PlayerModels_Server] Loaded {{{NbItems}}} items""");
}

Boolean Private_CurrentMapRequiresUnlock() {
  return C_EnvironmentRequiresUnlock.exists(G_CurrentMapEnvironmentName);
}

K_ModelIdentResult Private_GetModelIdent(Text _ModelId, Text _SkinId, Boolean _PreferCustom) {
  // If not specified to use custom items, try to match an archetype to preserve the regular skins.
  if (!_PreferCustom) {
    declare Archetype = Private_GetLookupId(_ModelId, True);

    if (G_LoadedItemIdByLookupId.existskey(Archetype)) {
      return K_ModelIdentResult{
        ModelIdent = G_LoadedItemIdByLookupId[Archetype],
        IsArchetype = True
      };
    }
  }

  // If a skin ID is specified, try to load the item with it.
  if (_SkinId != "") {
    declare WithSkin = Private_GetLookupId(_ModelId, _SkinId);

    if (G_LoadedItemIdByLookupId.existskey(WithSkin)) {
      return K_ModelIdentResult{
        ModelIdent = G_LoadedItemIdByLookupId[WithSkin],
        IsArchetype = False
      };
    }
  }

  // Load the item with a basic skin.
  declare Item = Private_GetLookupId(_ModelId);

  if (G_LoadedItemIdByLookupId.existskey(Item)) {
    return K_ModelIdentResult{
      ModelIdent = G_LoadedItemIdByLookupId[Item],
      IsArchetype = False
    };
  }

  // Nothing was found, fallback to no custom model.
  return K_ModelIdentResult{
    ModelIdent = NullId,
    IsArchetype = False
  };
}

Void Private_AssignModel(CPlayer _Player, Text _ModelId, Text _SkinCategoryId, Text _SkinId, Boolean _PreferCustom) {
  declare Result = Private_GetModelIdent(_ModelId, _SkinId, _PreferCustom);

  declare netwrite Integer Net_LibPlayerModels_LastUpdate for _Player;
  declare netwrite Text Net_LibPlayerModels_ModelId for _Player;
  declare netwrite Text Net_LibPlayerModels_SkinCategoryId for _Player;
  declare netwrite Text Net_LibPlayerModels_SkinId for _Player;
  declare netwrite Boolean Net_LibPlayerModels_PreferCustom for _Player;

  Net_LibPlayerModels_LastUpdate = G_Mode.Now;
  Net_LibPlayerModels_ModelId = _ModelId;
  Net_LibPlayerModels_SkinCategoryId = _SkinCategoryId;
  Net_LibPlayerModels_PreferCustom = _PreferCustom;

  if (Result.IsArchetype) {
    // Purposely blank out the skin id when using archetypes to avoid confusion in the UI.
    Net_LibPlayerModels_SkinId = "";
  } else {
    Net_LibPlayerModels_SkinId = _SkinId;
  }

  switchtype (_Player) {
    case CSmPlayer: {
      (_Player as CSmPlayer).ForceModelId = Result.ModelIdent;
    }
    case CTmPlayer: {
      (_Player as CTmPlayer).ForceModelId = Result.ModelIdent;
    }
    default: {
      assert(False, "Attempted to assign player model to unsupported player class instance.");
    }
  }
}

Void Unload() {
  G_Mode = Null;
  G_Options = K_Options {};
  G_SkinIdAllowlist.clear();
  G_LoadedItemIdByLookupId.clear();
}

Void Load(CMode _Mode, K_Options _Options) {
  Unload();

  G_Mode <=> _Mode;
  G_Options = _Options;

  CustomSkins::Load(G_Mode);

  Private_LoadModelItems();
}

Void Load(CMode _Mode) {
  Load(_Mode, K_Options {});
}

Void SetMapProperties(Text _Environment, Text _MapPlayerModelName) {
  G_CurrentMapEnvironmentName = _Environment;
  G_CurrentMapPlayerModelId = PMData::ResolveModelId(_MapPlayerModelName);
}

Text GetMapPlayerModelId() {
  return G_CurrentMapPlayerModelId;
}

/**
 * Get the player skin preference or a randomized skin (of a specified category).
 */
Text GetPlayerSkinId(CUser _User, Text _ResolvableModelId, Text _SkinCategoryId) {
  declare ModelId = PMData::ResolveModelId(_ResolvableModelId);

  return Private_GetPlayerSkinId(_User, ModelId, _SkinCategoryId);
}

/**
 * Get the player skin preference or a randomized skin.
 */
Text GetPlayerSkinId(CUser _User, Text _ResolvableModelId) {
  return GetPlayerSkinId(_User, _ResolvableModelId, PMData::C_DefaultSkinCategoryId);
}

/**
 * Resolve player skin preference or a randomized skin, directly or via a category.
 */
K_SkinIdResult ResolvePlayerSkinId(CUser _User, Text _ResolvableModelId, Text _ResolvableSkinId) {
  declare Result = PMData::ResolveSkinId(_ResolvableSkinId);

  // If a category is found, get the preference within the category.
  if (Result.SkinCategoryId != "") {
    return K_SkinIdResult{
      SkinCategoryId = Result.SkinCategoryId,
      SkinId = GetPlayerSkinId(_User, _ResolvableModelId, Result.SkinCategoryId)
    };
  }

  // If skin is found and directly called, assume the category as default.
  if (Result.SkinId != "") {
    return K_SkinIdResult{
      SkinCategoryId = PMData::C_DefaultSkinCategoryId,
      SkinId = Result.SkinId
    };
  }

  // Fallback to default preference.
  return K_SkinIdResult{
    SkinCategoryId = PMData::C_DefaultSkinCategoryId,
    SkinId = GetPlayerSkinId(_User, _ResolvableModelId)
  };
}

/**
 * Check if a model requires special unlock.
 */
Boolean GetModelRequiresUnlock(Text _ResolvableModelId) {
  declare ModelId = PMData::ResolveModelId(_ResolvableModelId);
  declare Archetype = PMData::GetModelArchetype(ModelId);

  return !C_VehicleIgnoresUnlock.exists(Archetype);
}

/**
 * Check if the player model should override an archetype.
 */
Boolean GetPlayerPreferCustom(CUser _User, Text _ResolvableModelId) {
  declare ModelId = PMData::ResolveModelId(_ResolvableModelId);

  if (CustomSkins::GetPlayerUseCustomOverArchetype(_User, ModelId)) {
    return True;
  }

  return Private_CurrentMapRequiresUnlock() && GetModelRequiresUnlock(ModelId);
}

/**
 * Get a player's target model ident.
 * For ghosts and bots only. For players, use AssignModel.
 */
Ident GetModelIdent(CPlayer _Player, Text _ResolvableModelId, Text _ResolvableSkinId) {
  declare ModelId = PMData::ResolveModelId(_ResolvableModelId);
  declare Result = ResolvePlayerSkinId(_Player.User, ModelId, _ResolvableSkinId);
  declare PreferCustom = _ResolvableSkinId != "" || GetPlayerPreferCustom(_Player.User, ModelId);

  return Private_GetModelIdent(ModelId, Result.SkinId, PreferCustom).ModelIdent;
}

/**
 * Get a player's target model ident (automatic skin).
 * For ghosts and bots only. For players, use AssignModel.
 */
Ident GetModelIdent(CPlayer _Player, Text _ResolvableModelId) {
  return GetModelIdent(_Player, _ResolvableModelId, "");
}

/**
 * Get a player's target model ident (automatic model from current map, automatic skin).
 * For ghosts and bots only. For players, use AssignModel.
 */
Ident GetModelIdent(CPlayer _Player) {
  return GetModelIdent(_Player, G_CurrentMapPlayerModelId);
}

/**
 * Assign a model to a player.
 */
Void AssignModel(CPlayer _Player, Text _ResolvableModelId, Text _ResolvableSkinId) {
  declare ModelId = PMData::ResolveModelId(_ResolvableModelId);
  declare Result = ResolvePlayerSkinId(_Player.User, ModelId, _ResolvableSkinId);
  declare PreferCustom = _ResolvableSkinId != "" || GetPlayerPreferCustom(_Player.User, ModelId);

  Private_AssignModel(_Player, ModelId, Result.SkinCategoryId, Result.SkinId, PreferCustom);
}

/**
 * Assign a model to a player (automatic skin).
 */
Void AssignModel(CPlayer _Player, Text _ResolvableModelId) {
  AssignModel(_Player, _ResolvableModelId, "");
}

/**
 * Assign a model to a player (automatic model from current map, automatic skin).
 */
Void AssignModel(CPlayer _Player) {
  AssignModel(_Player, G_CurrentMapPlayerModelId);
}
