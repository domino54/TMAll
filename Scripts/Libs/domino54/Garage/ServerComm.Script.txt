Void SetPreviewModelId(CUser _User, Text _ModelId) {
  declare Text LibGarage_CurrentPreviewModelId for _User;

  LibGarage_CurrentPreviewModelId = _ModelId;
}

Text GetPreviewModelId(CUser _User) {
  declare Text LibGarage_CurrentPreviewModelId for _User;

  return LibGarage_CurrentPreviewModelId;
}

Void SetPreviewSkinId(CUser _User, Text _SkinId) {
  declare Text LibGarage_CurrentPreviewSkinId for _User;

  LibGarage_CurrentPreviewSkinId = _SkinId;
}

Text GetPreviewSkinId(CUser _User) {
  declare Text LibGarage_CurrentPreviewSkinId for _User;

  return LibGarage_CurrentPreviewSkinId;
}

Void HideBackground(CUser _User, Integer _Delay) {
  declare Integer LibGarage_HideBackgroundAfter for _User;

  LibGarage_HideBackgroundAfter = Now + _Delay;
}

Void HideBackground(CUser _User) {
  HideBackground(_User, Now);
}

Boolean IsHideBackground(CUser _User, Integer _Time) {
  declare Integer LibGarage_HideBackgroundAfter for _User;

  if (LibGarage_HideBackgroundAfter <= 0) {
    return False;
  }

  declare Result = _Time >= LibGarage_HideBackgroundAfter;

  if (Result) {
    LibGarage_HideBackgroundAfter = -1;
  }

  return Result;
}
