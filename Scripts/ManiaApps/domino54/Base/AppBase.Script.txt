/**
 * yield; the script.
 * This should be the only place in the complete ManiaApp where the script yields.
 */
Void AB_Yield() {
  +++AB_Yield+++

  // yield after the loop, because some events may be otherwise missed.
  yield;
}

/**
 * Sleep the script.
 *
 * @param _Duration How long to sleep the script.
 */
Void AB_Sleep(Integer _Duration) {
  declare End = Now + _Duration;

  while (Now < End) {
    AB_Yield();
  }
}

// ---------------------------------- //
// Main
// ---------------------------------- //
main() {
  LayersDefaultManialinkVersion = 3;
  EnableMenuNavigationInputs = False;

  // Settings
  +++AB_Settings+++

  // Load libraries
  +++AB_LoadLibraries+++

  +++AB_LogVersions+++
  Log::DisplayScripts();

  // Init app and config
  +++AB_InitApp+++

  // Start app
  +++AB_StartApp+++

  while (True) {
    // Loop before the yield, because some events may be otherwise missed.
    +++AB_AppLoop+++

    AB_Yield();
  }

  // Stop app and cleanup
  +++AB_EndApp+++

  // Unload libraries
  +++AB_UnloadLibraries+++
}
