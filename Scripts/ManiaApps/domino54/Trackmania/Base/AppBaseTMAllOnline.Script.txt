if (_SplitScreenNum == 0) {
  declare Model = PlayerModels_Data::GetModelById(_ModelId);
  declare Skin = PlayerModels_Data::GetSkinById(_SkinId);
  declare ImageUrl = PlayerModels_Data::GetSkinAsset(_SkinId, GC::C_GarageAssetId_SkinCardIcon, "");

  if (ImageUrl == "") {
    ImageUrl = PlayerModels_Data::GetModelAsset(_ModelId, GC::C_GarageAssetId_SkinCardIcon, HC::C_ImgBase^GC::C_DefaultSkinCardIconUrl);
  }

  MenuOnline::SetModelName(Model.Name);
  MenuOnline::SetSkinName(Skin.Name);
  MenuOnline::SetModelImageUrl(ImageUrl);
}
