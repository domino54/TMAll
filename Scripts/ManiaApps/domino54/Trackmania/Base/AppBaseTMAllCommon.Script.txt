***AB_AppLoop***
***
foreach (Event in PendingEvents) {
  if (Event.Type != CManiaAppEvent::EType::LayerCustomEvent) continue;

  switch (Event.CustomEventType) {
    case PlayerModelsIngame::C_CustomEventType_PlayerUpdate: {
      if (Event.CustomEventData.count < 5) continue;

      declare SplitScreenNum = TL::ToInteger(Event.CustomEventData[0]);
      declare ModelId = Event.CustomEventData[1];
      declare SkinCategoryId = Event.CustomEventData[2];
      declare SkinId = Event.CustomEventData[3];

      TMAll_Private_ModelChange(SplitScreenNum, ModelId, SkinCategoryId, SkinId);
    }
  }
}

Void TMAll_Private_ModelChange(Integer _SplitScreenNum, Text _ModelId, Text _SkinCategoryId, Text _SkinId) {
  if (_SplitScreenNum == 0 && _ModelId != "") {
    // Set the model as the last viewed in the Garage.
    GaragePersistentAPI::SetLastModelId(_ModelId);
    GaragePersistentAPI::SetLastSkinCategoryId(_SkinCategoryId);
  }

  +++TMAll_OnModelChange+++
}
