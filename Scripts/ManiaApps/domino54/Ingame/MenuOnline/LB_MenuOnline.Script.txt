    ReturnValue = "\"\"\"" ^ _ModeName ^ "\"\"\"";
  }

  return """
Text GetModeName() {
  return {{{ReturnValue}}};
}
  """;
}

Integer GetMedalForTime(Integer _Time) {
  if (_Time <= 0) return 4;
  if (_Time <= Map.TMObjective_AuthorTime)  return 0;
  if (_Time <= Map.TMObjective_GoldTime)    return 1;
  if (_Time <= Map.TMObjective_SilverTime)  return 2;
  if (_Time <= Map.TMObjective_BronzeTime)  return 3;
  return 4;
}

Void SetBigMedal(Integer _Medal) {
  declare Quad_Medal <=> (Page.MainFrame.GetFirstChild("Quad_Medal") as CMlQuad);

  Quad_Medal.Visible = _Medal < 4 && C_QuadMedalImgColor.existskey(_Medal);

  if (Quad_Medal.Visible) {
    Quad_Medal.Colorize = C_QuadMedalImgColor[_Medal];
  }
}

Void SetTime(Integer _LastTime, Integer _BestTime) {
  declare LastMedal = GetMedalForTime(_LastTime);
  declare BestMedal = GetMedalForTime(_BestTime);

  declare PlayersName = [_("Last Time"), _("Best Time")];
  declare PlayersTime = ["-:--.---", "-:--.---"];
  declare PlayersColor = [C_QuadMedalImgColor[LastMedal], C_QuadMedalImgColor[BestMedal]];

  if (_LastTime > 0) {
    PlayersTime[0] = TimeToText(_LastTime);
  }

  if (_BestTime > 0) {
    PlayersTime[1] = TimeToText(_BestTime);
  }

  SetBigMedal(BestMedal);
  SetTimeInfo_SoloMapInfosBase(PlayersName, PlayersTime, PlayersColor);
}

Void SetModeName(Text _ModeName, Text _ServerName) {
  declare Label_Mode <=> (Page.MainFrame.GetFirstChild("Label_Mode") as CMlLabel);
  declare Label_Mode2 <=> (Page.MainFrame.GetFirstChild("Label_Mode2") as CMlLabel);
  declare Quad_Barre <=> (Page.MainFrame.GetFirstChild("Quad_Barre") as CMlQuad);

  if (C_PredefiendModeNames.existskey(_ModeName)) {
    Label_Mode.SetText(C_PredefiendModeNames[_ModeName]);
  } else {
    Label_Mode.SetText(_ModeName);
  }

  Label_Mode2.SetText(_ServerName);
  Quad_Barre.Show();
}

Void SetMapInfo(Text _MapName, Text _AuthorName, Text _CountryFlagUrl) {
  declare Label_TitleBody <=> (Page.GetFirstChild("Label_TitleBody") as CMlLabel);
  declare Label_Author <=> (Page.GetFirstChild("Label_Author") as CMlLabel);
  declare Quad_Flag <=> (Page.GetFirstChild("Quad_Flag") as CMlQuad);

  declare AuthorNickName = TL::Compose(_("By %1"), _AuthorName);
  declare LabelWidth = Label_Author.ComputeWidth(AuthorNickName);
  declare Flag = _CountryFlagUrl;

  if (Flag == "file://ZoneFlags/Path/") {
    Flag = "file://ZoneFlags/Path/World";
  }

  Label_TitleBody.SetText(_MapName);
  Label_Author.SetText(AuthorNickName);
  Quad_Flag.RelativePosition.X = LabelWidth - 32.0;
  Quad_Flag.ChangeImageUrl(Flag);
}

Void SendInfo(Boolean _IsSpectating) {
  declare Text[] LabelMedals = [
    _("Resume"),
    _("Help"),
    _("Spectator"),
    _("Advanced options"),
    _("Manage server"),
    _("Quit")
  ];

  declare Text SpectatingIcon;

  if (_IsSpectating) {
    // This is known in the professional environment as "padding"
    SpectatingIcon = "                       ";
  }

  SetButtons(LabelMedals, ["", "", SpectatingIcon], [], []);
}

Void Init() {
  SendInfo(IsSpectatorClient);
}

Void SetUIHidden(Boolean _Hidden) {
  ClientUI.AltMenuNoDefaultScores = _Hidden;
  ClientUI.AltMenuNoCustomScores = _Hidden;
  ClientUI.OverlayHideChat = _Hidden;

  if (_Hidden) {
    ClientUI.ScoreTableVisibility = CUIConfig::EVisibility::ForcedHidden;
  } else {
    ClientUI.ScoreTableVisibility = CUIConfig::EVisibility::None;
  }
}

Void ShowPage() {
  SetUIHidden(True);

  // CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::Resume);
  EnableMenuNavigation(True, False, Null, 2);

  G_IgnoreInputsUntil = Now + 100;

  AnimShow_QuadSelect();
  AnimShow_SoloMapInfosBase();

  //ClearChoose();
  FocusButton();
}

Void HidePage() {
  SetUIHidden(False);

  EnableMenuNavigation(False, False, Null, 0);
}

Void Button_Resume() {
  CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::Resume);
}

Void Button_Help() {
  ShowModeHelp();
}

Void Button_Spectator() {
  RequestSpectatorClient(!IsSpectatorClient);
}

Void Button_AdvancedMenu() {
  CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::AdvancedMenu);
}

Void Button_ServerSettings() {
  CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::ServerSettings);
}

Void Button_Quit() {
  CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::Quit);
}

Void PressSelect() {
  switch (G_ButtonFocus) {
    case 0: {
      Button_Resume();
    }
    case 1: {
      Button_Help();
    }
    case 2: {
      Button_Spectator();
    }
    case 3: {
      Button_AdvancedMenu();
    }
    case 4: {
      Button_ServerSettings();
    }
    case 5: {
      Button_Quit();
    }
  }
}

main() {
  Init();

  {{{Inject_SoloMapInfosBase::InjectInMain()}}}

  declare Boolean Prev_Visibility;
  declare Boolean Prev_IsSpectatorClient;

  declare PrevLastTime = -1;
  declare PrevBestTime = -1;

  if (InputPlayer != Null && InputPlayer.Score != Null) {
    PrevLastTime = InputPlayer.Score.PrevRace.Time;
    PrevBestTime = InputPlayer.Score.BestRace.Time;
  }

  SetTime(PrevLastTime, PrevBestTime);

  declare Prev_ModeName = GetModeName();
  declare Prev_ServerName = Playground.ServerInfo.ServerName;

  SetModeName(Prev_ModeName, Prev_ServerName);

  declare Text Prev_MapUid;

  if (Map != Null) {
    Prev_MapUid = Map.MapInfo.MapUid;

    SetMapInfo(Map.MapName, Map.AuthorNickName, Map.AuthorZoneIconUrl);
  }

  while (True) {
    yield;

    if (Prev_Visibility != PageIsVisible) {
      if (PageIsVisible) {
        ShowPage();
      } else {
        HidePage();
      }

      Prev_Visibility = PageIsVisible;
    }

    if (!PageIsVisible) {
      continue;
    }

    // Don't inject, it changes the big medal
    // Inject_SoloMapInfosBase::InjectInWhile()

    DesignManialinks_Loop();
    LibManialink_AnimLoop();

    {{{Inject_QuadsSelector::InjectInWhile("PressSelect();")}}}

    if (InputPlayer != Null && InputPlayer.Score != Null && PrevLastTime != InputPlayer.Score.PrevRace.Time || PrevBestTime != InputPlayer.Score.BestRace.Time) {
      PrevLastTime = InputPlayer.Score.PrevRace.Time;
      PrevBestTime = InputPlayer.Score.BestRace.Time;

      SetTime(PrevLastTime, PrevBestTime);
    }

    if (Prev_IsSpectatorClient != IsSpectatorClient) {
      Prev_IsSpectatorClient = IsSpectatorClient;

      SendInfo(IsSpectatorClient);
    }

    // Map has changed
    if (Map != Null && Prev_MapUid != Map.MapInfo.MapUid) {
      Prev_MapUid = Map.MapInfo.MapUid;

      SetMapInfo(Map.MapName, Map.AuthorNickName, Map.AuthorZoneIconUrl);
    }

    // Should only be used on online servers
    if (Prev_ModeName != GetModeName() || Prev_ServerName != Playground.ServerInfo.ServerName) {
      Prev_ModeName = GetModeName();
      Prev_ServerName = Playground.ServerInfo.ServerName;

      SetModeName(Prev_ModeName, Prev_ServerName);
    }

    foreach (Event in PendingEvents) {
      switch (Event.Type) {
        case CMlScriptEvent::Type::MenuNavigation: {
          switch (Event.MenuNavAction) {
            case CMlScriptEvent::EMenuNavAction::Cancel: {
              Button_Resume();
            }
          }
        }

        case CMlScriptEvent::Type::KeyPress: {
          if (Event.KeyName == "F1") {
            CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::NormalMenu);
          }
        }
      }
    }

    foreach (Event in Input.PendingEvents) {
      if (Now < G_IgnoreInputsUntil || Event.IsAutoRepeat) continue;

      switch (Event.Button) {
        case CInputEvent::EButton::Menu: {
          Button_Resume();
        }
        case CInputEvent::EButton::View: {
          CloseInGameMenu(CMlScriptIngame::EInGameMenuResult::NormalMenu);
        }
      }
    }
  }
}
