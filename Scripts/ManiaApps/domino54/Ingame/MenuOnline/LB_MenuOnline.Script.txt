}

Boolean CanDisplayMedals() {
  if (Map == Null || !Map.TMObjective_IsLapRace) {
    return True;
  }

  return !IndependantLaps && NbLaps == Map.TMObjective_NbLaps;
}

// TODO: Move this to Medals.Script.txt
Integer GetMedalForTime(Integer _Time) {
  if (_Time <= 0 || !CanDisplayMedals()) return 0;
  if (_Time <= Map.TMObjective_AuthorTime)  return 4;
  if (_Time <= Map.TMObjective_GoldTime)    return 3;
  if (_Time <= Map.TMObjective_SilverTime)  return 2;
  if (_Time <= Map.TMObjective_BronzeTime)  return 1;
  return 0;
}

Integer GetFastestTime(Integer[] _Times) {
  declare FastestTime = -1;

  foreach (Time in _Times) {
    if (Time > 0 && (FastestTime <= 0 || Time < FastestTime)) {
      FastestTime = Time;
    }
  }

  return FastestTime;
}

Void SetTime(Integer _LastTime, Integer _BestTime) {
  declare Frame_MenuCurrentMap <=> (Page.GetFirstChild("Frame_MenuCurrentMap") as CMlFrame);
  declare Frame_ScoreRows <=> (Page.GetFirstChild("Frame_ScoreRows") as CMlFrame);

  declare LastMedal = GetMedalForTime(_LastTime);
  declare BestMedal = GetMedalForTime(_BestTime);
  declare FastestTime = GetFastestTime([_LastTime, _BestTime]);

  declare PlayerNames = [_("Last Time"), _("Best Time")];
  declare PlayerTimes = ["-:--.---", "-:--.---"];
  declare PlayerMedals = [LastMedal, BestMedal];
  declare Boolean[] PlayerHighlight;

  if (_LastTime > 0) {
    PlayerTimes[0] = TimeToText(_LastTime);
  }

  if (_BestTime > 0) {
    PlayerTimes[1] = TimeToText(_BestTime);
  }

  PlayerHighlight.add(FastestTime > 0 && _LastTime == FastestTime);
  PlayerHighlight.add(FastestTime > 0 && _BestTime == FastestTime);

  HorizonCurrentMap_SetMedal(Frame_MenuCurrentMap, BestMedal);
  HorizonScoreRows_SetScores(Frame_ScoreRows, PlayerNames, PlayerTimes, PlayerMedals, PlayerHighlight);
}

Void SetSpectatorCheckbox(CMlFrame _Frame, Boolean _IsSpectating) {
  declare Quad_Checkbox <=> (_Frame.GetFirstChild("Quad_Checkbox") as CMlQuad);

  Quad_Checkbox.Visible = True;

  if (_IsSpectating) {
    Quad_Checkbox.ImageUrl = "{{{HC::C_ImgBase}}}Menus/Buttons/Checkbox_Closed.dds";
  } else {
    Quad_Checkbox.ImageUrl = "{{{HC::C_ImgBase}}}Menus/Buttons/Checkbox_Open.dds";
  }
}

Void SetUIHidden(Boolean _Hidden) {
  ClientUI.AltMenuNoDefaultScores = _Hidden;
  ClientUI.AltMenuNoCustomScores = _Hidden;
  //ClientUI.OverlayHideChat = _Hidden;

  if (_Hidden) {
    ClientUI.ScoreTableVisibility = CUIConfig::EVisibility::ForcedHidden;
  } else {
    ClientUI.ScoreTableVisibility = CUIConfig::EVisibility::None;
  }
}

Void ShowPage() {
  SetUIHidden(True);

  EnableMenuNavigation(True, False, Null, 2);

  HorizonHeader_Show((Page.GetFirstChild("Frame_MenuHeader") as CMlFrame));
  HorizonDualFrame_Show((Page.GetFirstChild("Frame_MenuCurrentMap") as CMlFrame));
  HorizonDualFrame_Show((Page.GetFirstChild("Frame_MenuPlayerModel") as CMlFrame));
  HorizonMenuOption_Show((Page.GetFirstChild("Frame_MenuOptions") as CMlFrame));

  HorizonFocus_FocusInitial();
}

Void HidePage() {
  SetUIHidden(False);

  EnableMenuNavigation(False, False, Null, 0);
}

Void SetButtons(Boolean _UseClans) {
  declare Frame_MenuPlayerModel <=> (Page.GetFirstChild("Frame_MenuPlayerModel") as CMlFrame);
  declare Frame_MenuOptions <=> (Page.GetFirstChild("Frame_MenuOptions") as CMlFrame);
  declare Frame_BottomLine <=> (Page.GetFirstChild("Frame_BottomLine") as CMlFrame);

  declare Text[] ButtonLabels;
  declare Text[] MenuOptions;

  ButtonLabels.add(_("Resume"));
  MenuOptions.add(C_ButtonResume);

  ButtonLabels.add(_("Help"));
  MenuOptions.add(C_ButtonHelp);

  ButtonLabels.add(_("Spectator"));
  MenuOptions.add(C_ButtonSpectator);

  if (_UseClans) {
    ButtonLabels.add(_("Change team"));
    MenuOptions.add(C_ButtonChangeTeam);
  }

  ButtonLabels.add(_("Advanced options"));
  MenuOptions.add(C_ButtonAdvancedMenu);

  ButtonLabels.add(_("Manage server"));
  MenuOptions.add(C_ButtonServerSetting);

  ButtonLabels.add(_("Quit"));
  MenuOptions.add(C_ButtonQuit);

  HorizonFocus_NavigationStackAdd(Frame_MenuPlayerModel, C_ButtonGarage);

  foreach (I => Control in Frame_MenuOptions.Controls) {
    Control.Visible = ButtonLabels.existskey(I);

    if (!Control.Visible) {
      continue;
    }

    HorizonFocus_NavigationStackAdd(Control, MenuOptions[I]);
    HorizonMenuOption_SetText((Control as CMlFrame), ButtonLabels[I]);
  }

  HorizonFocus_SetInitialControl(Frame_MenuOptions.Controls[0]);

  Frame_BottomLine.RelativePosition_V3.Y = -146.0;

  if (_UseClans) {
    Frame_BottomLine.RelativePosition_V3.Y -= {{{HC::C_MenuOptionHeight + HC::C_MenuOptionSpacing}}};
  }
}

main() {
  declare Frame_MenuHeader <=> (Page.GetFirstChild("Frame_MenuHeader") as CMlFrame);
  declare Frame_MenuCurrentMap <=> (Page.GetFirstChild("Frame_MenuCurrentMap") as CMlFrame);
  declare Frame_MenuPlayerModel <=> (Page.GetFirstChild("Frame_MenuPlayerModel") as CMlFrame);
  declare Frame_MenuOptions <=> (Page.GetFirstChild("Frame_MenuOptions") as CMlFrame);
  declare Frame_SpectatorOption <=> (Frame_MenuOptions.Controls[2] as CMlFrame);

  declare Text CurrentModelName for Page;
  declare Text CurrentSkinName for Page;
  declare Text CurrentModelImageUrl for Page;
  declare UpdatePlayerModel = False;

  HorizonCurrentModel_SetInfo(Frame_MenuPlayerModel, CurrentModelName, CurrentSkinName, CurrentModelImageUrl);

  declare Boolean PrevIsPageVisible;
  declare Boolean Prev_IsSpectatorClient;

  SetSpectatorCheckbox(Frame_SpectatorOption, IsSpectatorClient);

  declare PrevLastTime = -1;
  declare PrevBestTime = -1;

  if (InputPlayer != Null && InputPlayer.Score != Null) {
    if (InputPlayer.Score.PrevRace != Null) {
      PrevLastTime = InputPlayer.Score.PrevRace.Time;
    }
    if (InputPlayer.Score.BestRace != Null) {
      PrevBestTime = InputPlayer.Score.BestRace.Time;
    }
  }

  SetTime(PrevLastTime, PrevBestTime);

  declare PrevModeName = GetModeName();
  declare PrevServerName = Playground.ServerInfo.ServerName;

  HorizonHeader_SetTitle(Frame_MenuHeader, PrevModeName);
  HorizonHeader_SetSubtitle(Frame_MenuHeader, Playground.ServerInfo.ServerName);

  declare Text PrevMapUid;
  declare PrevCanChangeTeam = False;

  SetButtons(False);

  +++Main+++

  while (True) {
    {{{HorizonFocus::InjectLoop("PressSelect(Value);")}}}

    foreach (Event in PendingEvents) {
      if (Event.Type != CMlScriptEvent::Type::PluginCustomEvent) continue;
      if (Event.CustomEventType != {{{dump(C_CustomEventType_ModelUpdate)}}}) continue;

      UpdatePlayerModel = True;
    }

    if (PrevIsPageVisible != PageIsVisible) {
      if (PageIsVisible) {
        ShowPage();
      } else {
        HidePage();
      }

      PrevIsPageVisible = PageIsVisible;
    }

    if (!PageIsVisible) {
      yield;
      continue;
    }

    if (InputPlayer != Null && InputPlayer.Score != Null) {
      if (
        (InputPlayer.Score.PrevRace != Null && PrevLastTime != InputPlayer.Score.PrevRace.Time) ||
        (InputPlayer.Score.BestRace != Null && PrevBestTime != InputPlayer.Score.BestRace.Time)
      ) {
        PrevLastTime = InputPlayer.Score.PrevRace.Time;
        PrevBestTime = InputPlayer.Score.BestRace.Time;

        SetTime(PrevLastTime, PrevBestTime);
      }
    }

    if (Prev_IsSpectatorClient != IsSpectatorClient) {
      Prev_IsSpectatorClient = IsSpectatorClient;

      SetSpectatorCheckbox(Frame_SpectatorOption, IsSpectatorClient);
    }

    // Map has changed
    if (Map != Null && PrevMapUid != Map.MapInfo.MapUid) {
      PrevMapUid = Map.MapInfo.MapUid;

      HorizonCurrentMap_SetMapInfo(Frame_MenuCurrentMap, Map.MapName, Map.AuthorNickName, Map.AuthorZoneIconUrl);
    }

    // Should only be used on online servers
    if (PrevModeName != GetModeName()) {
      PrevModeName = GetModeName();

      HorizonHeader_SetTitle(Frame_MenuHeader, PrevModeName);
    }

    if (PrevServerName != Playground.ServerInfo.ServerName) {
      PrevServerName = Playground.ServerInfo.ServerName;

      HorizonHeader_SetSubtitle(Frame_MenuHeader, Playground.ServerInfo.ServerName);
    }

    // Team change is possible
    declare CanChangeTeam = !IsSpectator && UseClans && !UseForcedClans;

    if (PrevCanChangeTeam != CanChangeTeam) {
      PrevCanChangeTeam = CanChangeTeam;

      SetButtons(CanChangeTeam);
    }

    if (UpdatePlayerModel) {
      UpdatePlayerModel = False;

      HorizonCurrentModel_SetInfo(Frame_MenuPlayerModel, CurrentModelName, CurrentSkinName, CurrentModelImageUrl);
    }

    foreach (Event in PendingEvents) {
      if (Event.Type != CMlScriptEvent::Type::MenuNavigation) continue;

      switch (Event.MenuNavAction) {
        case CMlScriptEvent::EMenuNavAction::Cancel: {
          Button_Resume();
        }
        case CMlScriptEvent::EMenuNavAction::Action2: {
          Button_NormalMenu();
        }
      }
    }

    +++Yield+++

    yield;
  }
}
