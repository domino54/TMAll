#Include "ManiaApps/Nadeo/Layers.Script.txt" as Layers

// ---------------------------------- //
// Global variables
// ---------------------------------- //

declare Text[] G_EditorUILayerIds;

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void AddLayer(Text _LayerId) {
  if (G_EditorUILayerIds.exists(_LayerId)) {
    return;
  }

  G_EditorUILayerIds.add(_LayerId);
}

Void SetVisibility(Boolean _IsVisible) {
  HideEditorInterface = !_IsVisible;

  foreach (LayerId in G_EditorUILayerIds) {
    Layers::SetVisibility(LayerId, !HideEditorInterface);
  }
}
