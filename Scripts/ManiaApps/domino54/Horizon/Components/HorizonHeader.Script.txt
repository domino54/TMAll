Text CreateHeader(Text _Id, Real _MaxWidth, Text _Title, Text _Subtitle) {
  declare MainFont = Font::GetFontName("Main");
  declare LabelMaxWidth = _MaxWidth - HC::C_HeaderLineWidth - HC::C_HeaderLabelMargin;
  declare LabelStyles = """ valign="center2" size="{{{LabelMaxWidth}}}" textemboss="1" textcolor="{{{HC::C_HeaderTextColor}}}" textfont="{{{MainFont}}}" """;

  return """
<frame id="{{{_Id}}}">
  <quad id="Quad_HeaderDecorLine" pos="0 {{{HC::C_HeaderFixedHeight / -2}}}" valign="center" size="{{{HC::C_HeaderLineWidth}}} {{{HC::C_HeaderFixedHeight}}}" bgcolor="{{{HC::C_HeaderTextColor}}}" />

  <frame pos="{{{HC::C_HeaderLineWidth}}} {{{HC::C_HeaderFixedHeight / -2}}}" valign="center" size="{{{_MaxWidth - HC::C_HeaderLineWidth}}} 90">
    <label id="Label_HeaderTitleText" pos="{{{HC::C_HeaderLabelMargin}}} 2.25" {{{LabelStyles}}} textsize="{{{HC::C_HeaderTextSizeTitle}}}" text="{{{_Title}}}" />
    <label id="Label_HeaderSubtitleText" pos="{{{HC::C_HeaderLabelMargin}}} -5.75" {{{LabelStyles}}} textsize="{{{HC::C_HeaderTextSizeSubtitle}}}" text="{{{_Subtitle}}}" />
  </frame>
</frame>
  """;
}

Text CreateHeader(Text _Id, Real _MaxWidth, Text _Title) {
  return CreateHeader(_Id, _MaxWidth, _Title, "");
}

Text CreateHeader(Text _Id, Real _MaxWidth) {
  return CreateHeader(_Id, _MaxWidth, "");
}

Text CreateHeader(Text _Id, Text _Title, Text _Subtitle) {
  return CreateHeader(_Id, HC::C_MenuOptionWidth, _Title, _Subtitle);
}

Text CreateHeader(Text _Id, Text _Title) {
  return CreateHeader(_Id, _Title, "");
}

Text CreateHeader(Text _Id) {
  return CreateHeader(_Id, "");
}

Text InjectFunctions() {
  return """// <script>

Void HorizonHeader_LaunchAnimation(CMlFrame _Frame, Boolean _IsVisible, Integer _Duration) {
  declare Quad_HeaderDecorLine <=> (_Frame.GetFirstChild("Quad_HeaderDecorLine") as CMlQuad);
  declare Label_HeaderTitleText <=> (_Frame.GetFirstChild("Label_HeaderTitleText") as CMlLabel);
  declare Label_HeaderSubtitleText <=> (_Frame.GetFirstChild("Label_HeaderSubtitleText") as CMlLabel);

  AnimMgr.Flush(Quad_HeaderDecorLine);
  AnimMgr.Flush(Label_HeaderTitleText);
  AnimMgr.Flush(Label_HeaderSubtitleText);

  declare Easing = CAnimManager::EAnimManagerEasing::QuintIn;
  declare PosTitle = {{{HC::C_HeaderLabelMargin}}};
  declare PosSubtitle = {{{HC::C_HeaderLabelMargin}}};
  declare LineHeight = {{{HC::C_HeaderFixedHeight}}};
  declare WidthTitle = ML::Min(Label_HeaderTitleText.Size.X, Label_HeaderTitleText.ComputeWidth(Label_HeaderTitleText.Value));
  declare WidthSubtitle = ML::Min(Label_HeaderSubtitleText.Size.X, Label_HeaderSubtitleText.ComputeWidth(Label_HeaderSubtitleText.Value));

  if (_IsVisible) {
    Easing = CAnimManager::EAnimManagerEasing::QuintOut;
    Quad_HeaderDecorLine.Size.Y = 0.0;
    Label_HeaderTitleText.RelativePosition_V3.X = -WidthTitle;
    Label_HeaderSubtitleText.RelativePosition_V3.X = -WidthSubtitle;
  } else {
    PosTitle = -WidthTitle;
    PosSubtitle = -WidthSubtitle;
    LineHeight = 0.0;
  }

  AnimMgr.Add(Quad_HeaderDecorLine, "<control size='"^Quad_HeaderDecorLine.Size.X^" "^LineHeight^"' />", _Duration, Easing);
  AnimMgr.Add(Label_HeaderTitleText, "<control pos='"^PosTitle^" "^Label_HeaderTitleText.RelativePosition_V3.Y^"' />", _Duration, Easing);
  AnimMgr.Add(Label_HeaderSubtitleText, "<control pos='"^PosSubtitle^" "^Label_HeaderSubtitleText.RelativePosition_V3.Y^"' />", _Duration, Easing);
}

Void HorizonHeader_Show(CMlFrame _Frame, Integer _Duration) {
  HorizonHeader_LaunchAnimation(_Frame, True, _Duration);
}

Void HorizonHeader_Show(CMlFrame _Frame) {
  HorizonHeader_Show(_Frame, {{{HC::C_HeaderAnimDuration}}});
}

Void HorizonHeader_Hide(CMlFrame _Frame, Integer _Duration) {
  HorizonHeader_LaunchAnimation(_Frame, False,_Duration);
}

Void HorizonHeader_Hide(CMlFrame _Frame) {
  HorizonHeader_Hide(_Frame, {{{HC::C_HeaderAnimDuration}}});
}

Void HorizonHeader_SetTitle(CMlFrame _Frame, Text _Title) {
  declare Label_HeaderTitleText <=> (_Frame.GetFirstChild("Label_HeaderTitleText") as CMlLabel);

  Label_HeaderTitleText.SetText(_Title);
}

Void HorizonHeader_SetSubtitle(CMlFrame _Frame, Text _Subtitle) {
  declare Label_HeaderSubtitleText <=> (_Frame.GetFirstChild("Label_HeaderSubtitleText") as CMlLabel);

  Label_HeaderSubtitleText.SetText(_Subtitle);
}

  """;
}
