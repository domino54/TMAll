Void SendModeEvent(Text _CustomEventName, Text[] _CustomEventData) {
  declare CNod Context = This;
  if (!(Context is CManiaAppPlayground)) return;
  (Context as CManiaAppPlayground).SendCustomEvent(_CustomEventName, _CustomEventData);
}

Integer GetGameTime() {
  declare CNod Context = This;
  if (!(Context is CManiaAppPlayground)) return Now;

}

Void CloseBrowser() {
  declare CNod Context = This;
  if (!(Context is CManiaAppBrowser)) return;

}

Void Close() {
  if (GaragePlaygroundAPI::IsPlayground()) {
    SendModeEvent(GarageServer::C_CustomEventType_StopServer, []);
  } else {
    CloseBrowser();
  }
}

K_Skin[] GetSelectableSkins(Text _ModelId, Text _SkinCategoryId) {
  declare K_Skin[] Result;

  // Add a default option for handling the unselected skin state.
  Result.add(K_Skin {
    Id = "",
    Name = _("(randomized)"),
    AppliesTo = _ModelId
  });

  // Add existing skin options.
  declare K_Skin[Text] SkinsToSort;
  declare Text[] ExistingSkinIds;

  foreach (Skin in PMData::GetSkinsListByModelId(_ModelId, _SkinCategoryId)) {
    SkinsToSort[TL::ToUpperCase("""{{{Skin.IsCustom}}}|{{{Skin.SortIndex}}}|{{{Skin.Name}}}|{{{Skin.Id}}}""")] = Skin;
  }

  SkinsToSort = SkinsToSort.sortkey();

  foreach (Skin in SkinsToSort) {
    Result.add(Skin);
    ExistingSkinIds.add(Skin.Id);
  }

  // If it doesn't exist in the data, add the current skin ID as a fallback option.
  declare PlayerSkinId = GetSelectedSkinId(_ModelId, _SkinCategoryId);

  if (PlayerSkinId != "" && !ExistingSkinIds.exists(PlayerSkinId)) {
    Result.add(K_Skin {
      Id = PlayerSkinId,
      Name = PlayerSkinId,
      AppliesTo = _ModelId
    });
  }

  return Result;
}

Text[] GetSelectableModelIds() {
  declare Text[] Result;

  foreach (Model in PMData::GetModelsList()) {
    Result.add(Model.Id);
  }

  return Result;
}

Text[] GetSelectableSkinCategoryIds() {
  declare Text[] Result;

  foreach (SkinCategory in PMData::GetAllSkinCategories()) {
    Result.add(SkinCategory.Id);
  }

  return Result;
}

Text[] GetSelectableSkinCategoryNames() {
  declare Text[] Result;

  foreach (SkinCategory in PMData::GetAllSkinCategories()) {
    Result.add(SkinCategory.Name);
  }

  return Result;
}

Text GetDefaultSkinCategoryId() {
  return PMData::C_DefaultSkinCategoryId;
}

Text GetInitialSkinCategoryId() {
  declare SkinCategoryIds = GetSelectableSkinCategoryIds();
  declare LastSkinCategoryId = GaragePersistentAPI::GetLastSkinCategoryId();

  if (SkinCategoryIds.exists(LastSkinCategoryId)) {
    return LastSkinCategoryId;
  }

  if (SkinCategoryIds.count > 0) {
    return SkinCategoryIds[0];
  }

  return GetDefaultSkinCategoryId();
}

Text[] GetSelectableSkinIds(Text _ModelId, Text _SkinCategoryId) {
  declare Text[] Result;

  foreach (Skin in GetSelectableSkins(_ModelId, _SkinCategoryId)) {
    Result.add(Skin.Id);
  }

  return Result;
}

Text GetInitialModelId() {
  declare ModelIds = GetSelectableModelIds();
  declare LastModelId = GaragePersistentAPI::GetLastModelId();

  if (ModelIds.exists(LastModelId)) {
    return LastModelId;
  }

  if (ModelIds.count > 0) {
    return ModelIds[0];
  }

  return "";
}

Text GetModelId() {
  return G_PreviewModelId;
}

Void SetModelId(Text _ModelId) {
  G_PreviewModelId = _ModelId;

  declare SkinCategoryId = GetSkinCategoryId();
  declare PlayerSkinId = GetSelectedSkinId(_ModelId, SkinCategoryId);
  declare SkinIds = GetSelectableSkinIds(_ModelId, SkinCategoryId);

  GaragePersistentAPI::SetLastModelId(_ModelId);
  GarageServer::SetPreviewModelId(LocalUser, _ModelId);

  if (SkinIds.exists(PlayerSkinId)) {
    SetSkinId(PlayerSkinId);
  }
  else if (SkinIds.count > 0) {
    SetSkinId(SkinIds[0]);
  }
  else {
    SetSkinId("");
  }
}
