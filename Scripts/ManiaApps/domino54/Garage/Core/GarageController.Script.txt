#Const Version      "2024-12-22"
#Const ScriptName   "ManiaApps/domino54/Garage/Core/GarageController.Script.txt"

#Include "ManiaApps/domino54/Garage/Components/GarageBackground/GarageBackground.Script.txt" as GarageBg
#Include "ManiaApps/domino54/Garage/Components/SkinPreview/SkinPreview.Script.txt" as SkinPreview
#Include "ManiaApps/domino54/Garage/Core/GarageAPI.Script.txt" as GarageAPI
#Include "ManiaApps/domino54/Garage/Core/GaragePlaygroundAPI.Script.txt" as GaragePlaygroundAPI
#Include "ManiaApps/domino54/Garage/Pages/GarageIntroPage.Script.txt" as GarageIntroPage
#Include "ManiaApps/domino54/Garage/Pages/GarageMainPage.Script.txt" as GarageMainPage
#Include "ManiaApps/domino54/Garage/Pages/GarageTutorialPage.Script.txt" as GarageTutorialPage

// ---------------------------------- //
// Global variables
// ---------------------------------- //

declare Text G_CurrentPage;

// ---------------------------------- //
// Private functions
// ---------------------------------- //

Void Private_HidePage(Text _PageName) {
  switch (_PageName) {
    case GarageIntroPage::C_PageName: GarageIntroPage::Hide();
    case GarageMainPage::C_PageName: GarageMainPage::Hide();
    case GarageTutorialPage::C_PageName: GarageTutorialPage::Hide();
  }
}

Void Private_ShowPage(Text _PageName) {
  switch (_PageName) {
    case GarageIntroPage::C_PageName: GarageIntroPage::Show();
    case GarageMainPage::C_PageName: GarageMainPage::Show();
    case GarageTutorialPage::C_PageName: GarageTutorialPage::Show();
  }
}

Void Private_PageLoop(Text _PageName) {
  switch (_PageName) {
    case GarageIntroPage::C_PageName: GarageIntroPage::Loop();
    case GarageMainPage::C_PageName: GarageMainPage::Loop();
    case GarageTutorialPage::C_PageName: GarageTutorialPage::Loop();
  }
}

Void Private_SetActivePage(Text _PageName) {
  Private_HidePage(G_CurrentPage);
  G_CurrentPage = _PageName;
  Private_ShowPage(G_CurrentPage);
}

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Init() {
  GarageBg::Load(GaragePlaygroundAPI::IsPlayground());
}

Void Load() {
  GarageIntroPage::Load();
  GarageMainPage::Load();
  GarageTutorialPage::Load();
  SkinPreview::Load();
}

Void Show() {
  if (GaragePlaygroundAPI::IsPlayground()) {
    GarageBg::Hide();
  } else {
    SkinPreview::Show();
  }

  declare InitialPageName = GarageIntroPage::C_PageName;

  if (GarageAPI::HasSeenTutorial(GarageTutorialPage::C_TutorialVersion)) {
    InitialPageName = GarageMainPage::C_PageName;
  }

  Private_SetActivePage(InitialPageName);
}

Void Loop() {
  Private_PageLoop(G_CurrentPage);
  SkinPreview::Loop();
}

Void Unload() {
  GarageIntroPage::Unload();
  GarageMainPage::Unload();
  GarageTutorialPage::Unload();
  GarageBg::Unload();
  SkinPreview::Unload();
}
