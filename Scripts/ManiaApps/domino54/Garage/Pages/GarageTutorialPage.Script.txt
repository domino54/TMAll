#Const Version      "2024-12-22"

#Include "ManiaApps/domino54/Garage/Components/TutorialScreen/TutorialScreen.Script.txt" as TutorialScreen
#Include "ManiaApps/domino54/Garage/Core/GarageRouter.Script.txt" as GarageRouter

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_PageName         "GarageTutorialPage"
#Const C_TutorialVersion  1

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Unload() {
  TutorialScreen::Unload();
}

Void Load() {
  Unload();

  TutorialScreen::Load();
}

Void Show() {
  TutorialScreen::Show();
}

Void Hide() {
  TutorialScreen::Hide();
}

Void Loop() {
  foreach (Event in PendingEvents) {
    if (Event.Type != CManiaAppEvent::EType::LayerCustomEvent) continue;

    if (Event.CustomEventType == "CloseTutorial") {
      GarageRouter::SetCurrentRoute("GarageMainPage");
    }
  }
}
