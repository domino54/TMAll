#Const Version      "2024-12-25"
#Const ScriptName   "ManiaApps/domino54/Garage/Pages/GarageMainPage.Script.txt"

#Include "ColorLib" as CL
#Include "ManiaApps/domino54/Garage/Components/ModelTabs/ModelTabs.Script.txt" as ModelTabs
#Include "ManiaApps/domino54/Garage/Components/SkinAuthorName/SkinAuthorName.Script.txt" as SkinAuthorName
#Include "ManiaApps/domino54/Garage/Components/SkinFallbackInfo/SkinFallbackInfo.Script.txt" as SkinFallbackInfo
#Include "ManiaApps/domino54/Garage/Components/SkinList/SkinList.Script.txt" as SkinList
#Include "ManiaApps/domino54/Garage/Components/TitleLogoControls/TitleLogoControls.Script.txt" as TitleLogoControls
#Include "ManiaApps/domino54/Garage/Core/GarageAPI.Script.txt" as GarageAPI

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_PageName     "GarageMainPage"
#Const C_GarageColor  "736AE6"  // TODO configurable

// ---------------------------------- //
// Structs
// ---------------------------------- //

#Struct ModelTabs::K_MenuTab as K_MenuTab
#Struct SkinList::K_SkinOption as K_SkinOption

// ---------------------------------- //
// Private functions
// ---------------------------------- //

Text Private_GetTitleText() {
  if (GarageAPI::IsVehicles()) {
    return _("Garage");
  }

  if (GarageAPI::IsCharacters()) {
    return _("Wardrobe");
  }

  return _("Models");
}

K_MenuTab[] Private_CreateModelTabs() {
  declare K_MenuTab[] Result;

  foreach (Model in GarageAPI::GetSelectableModels()) {
    Result.add(K_MenuTab {
      Id = Model.Id,
      Label = GarageAPI::GetModelName(Model.Id, Model.Name),
      IconUrl = GarageAPI::GetModelIconUrl(Model.Id)
    });
  }

  return Result;
}

K_SkinOption[] Private_CreateSkinOptions(Text _ModelId, Text _SkinCategoryId) {
  declare K_SkinOption[] Result;

  foreach (Skin in GarageAPI::GetSelectableSkins(_ModelId, _SkinCategoryId)) {
    Result.add(K_SkinOption {
      Id = Skin.Id,
      Name = Skin.Name,
      IconUrl = GarageAPI::GetSkinCardUrl(Skin)
    });
  }

  return Result;
}

Void Private_UpdateCurrentSkins() {
  declare ModelId = GarageAPI::GetModelId();
  declare SkinCategoryId = GarageAPI::GetSkinCategoryId();
  declare SkinId = GarageAPI::GetSkinId();

  SkinList::SetSkinsData(Private_CreateSkinOptions(ModelId, SkinCategoryId));
  SkinList::SetPreviewSkinId(SkinId);
  SkinList::SetAppliedSkinId(GarageAPI::GetSelectedSkinId(ModelId, SkinCategoryId));
  SkinAuthorName::SetAuthorName(GarageAPI::GetSkinAuthorName(SkinId));
  SkinFallbackInfo::SetButtonVisibility(GarageAPI::GetUsesArchetype(ModelId));
  SkinFallbackInfo::SetPreferCustom(GarageAPI::GetPreferCustom(ModelId));
}

Void Private_TogglePreferCustom(Text _ModelId) {
  declare PreferCustom = !GarageAPI::GetPreferCustom(_ModelId);

  GarageAPI::SetPreferCustom(_ModelId, PreferCustom);
  SkinFallbackInfo::SetPreferCustom(PreferCustom);
}

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Unload() {
  ModelTabs::Unload();
  SkinAuthorName::Unload();
  SkinFallbackInfo::Unload();
  SkinList::Unload();
  TitleLogoControls::Unload();
}

Void Load() {
  Unload();

  TitleLogoControls::Load();
  ModelTabs::Load();
  ModelTabs::SetHeaderText(Private_GetTitleText());
  ModelTabs::SetColor(CL::HexToRgb(C_GarageColor));
  SkinAuthorName::Load();
  SkinFallbackInfo::Load();
  SkinList::Load();

  ModelTabs::SetTabsData(Private_CreateModelTabs());
  ModelTabs::SetActiveTabId(GarageAPI::GetModelId());

  Private_UpdateCurrentSkins();
}

Void Show() {
  TitleLogoControls::Show();
  ModelTabs::Show();
  SkinAuthorName::Show();
  SkinFallbackInfo::Show();
  SkinList::Show();
}

Void Hide() {
  TitleLogoControls::Show();
  ModelTabs::Hide();
  SkinAuthorName::Show();
  SkinFallbackInfo::Show();
  SkinList::Show();
}

Void Loop() {
  foreach (Event in PendingEvents) {
    if (Event.Type != CManiaAppEvent::EType::LayerCustomEvent) continue;

    switch (Event.CustomEventType) {
      case ModelTabs::C_CustomEventType_ActiveTab: {
        GarageAPI::SetModelId(Event.CustomEventData[0]);

        Private_UpdateCurrentSkins();
      }
      case SkinList::C_CustomEventType_SetPreviewSkinId: {
        GarageAPI::SetSkinId(Event.CustomEventData[0]);

        SkinAuthorName::SetAuthorName(GarageAPI::GetSkinAuthorName(GarageAPI::GetSkinId()));
      }
      case SkinList::C_CustomEventType_SetAppliedSkinId: {
        GarageAPI::SetSelectedSkinId(GarageAPI::GetModelId(), GarageAPI::GetSkinCategoryId(), Event.CustomEventData[0]);
      }
      case SkinFallbackInfo::C_CustomEventType_TogglePreferCustom: {
        Private_TogglePreferCustom(GarageAPI::GetModelId());
      }
    }

    if (Event.CustomEventType == "GlobalB_Back") {
      GarageAPI::Close();
    }
  }
}
