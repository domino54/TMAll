<quad id="Quad_SkinPreview" pos="0 0" z-index="-5" halign="center" valign="center" size="180 180" keepratio="fit" autoscale="0" />

declare Quad_SkinPreview <=> (Page.GetFirstChild("Quad_SkinPreview") as CMlQuad);

while (True) {
  wait(PendingEvents.count > 0);

  foreach (Event in PendingEvents) {
    if (Event.Type != CMlScriptEvent::Type::PluginCustomEvent) continue;

    if (!Event.CustomEventData.existskey(0)) continue;

    Quad_SkinPreview.ChangeImageUrl(Event.CustomEventData[0]);
  }

  yield;
}

</manialink>
  """;
