#Const Version      "2024-10-30"

#Include "ManiaApps/Nadeo/Layers.Script.txt" as Layers
#Include "ManiaApps/domino54/Garage/Components/GarageBackground/LB_GarageBackground.Script.txt" as LB_GarageBackground

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_ManialinkLayerId "L_GarageBackground"

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void Unload() {
  Layers::Destroy(C_ManialinkLayerId);
}

Void Load(Boolean _SingleImage) {
  Unload();

  Layers::Create(C_ManialinkLayerId, LB_GarageBackground::Build(_SingleImage));
}

Void Hide() {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);

  declare Boolean HideBackground for Layer.LocalPage;
  HideBackground = True;
}
