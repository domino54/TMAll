#Const Version      "2024-12-25"
#Const ScriptName   "ManiaApps/domino54/Garage/Components/SkinFallbackInfo/LB_SkinFallbackInfo.Script.txt"

#Include "Libs/domino54/MouseExt.Script.txt" as MouseExt
#Include "Libs/Nadeo/MP4/Common/ManialinkTools/FontManager.Script.txt" as Font
#Include "ManiaApps/domino54/Common/DesignButtons.Script.txt" as DesignButtons

#Const C_LabelTitle         _("Fallback Skin")
#Const C_LabelDescription   _("This skin will only be used in special circumstances, where the profile skin cannot normally work.\nYou can also opt in to use this skin everywhere instead, which will override your profile selection.")
#Const C_LabelButton        _("Override Profile Skin")
#Const C_FramePosition      <-150.0, 51.0>
#Const C_FrameWidth         68.0
#Const C_FrameTextMargin    3.0
#Const C_ButtonHeight       8.0
#Const C_ButtonSpacingGap   2.0
#Const C_ForegroundColor    "222222"
#Const C_BackgroundColor    "FFFFFF"
#Const C_BackgroundOpacity  0.5

#Const C_CustomEventType_TogglePreferCustom   "skin_fallback_info:toggle_prefer_custom"
#Const C_CustomEventType_SetPreferCustom      "skin_fallback_info:set_prefer_custom"
#Const C_CustomEventType_SetIsVisible         "skin_fallback_info:set_is_visible"

Text Build() {
  declare SubFont = Font::GetFontName("SubTitle");
  declare TitleY = C_FrameTextMargin + 1.5;
  declare LabelWidth = C_FrameWidth - (C_FrameTextMargin * 2);
  declare ButtonSize = <C_FrameWidth, C_ButtonHeight>;
  declare HideOffset = -100.0;

  return """
<manialink version="3" name="L_SkinFallbackInfo">
<frame id="Frame_MoveContainer" pos="{{{C_FramePosition.X}}} {{{C_FramePosition.Y}}}">
  <frame id="Frame_HintWindow" pos="{{{HideOffset}}}">
    <label pos="{{{C_FrameTextMargin}}} {{{-TitleY}}}" valign="center2" textsize="2" textcolor="{{{C_ForegroundColor}}}" text="" />
    <label pos="{{{C_FrameTextMargin + 5}}} {{{-TitleY}}}" halign="left" valign="center2" textfont="{{{SubFont}}}" textsize="1.5" textcolor="{{{C_ForegroundColor}}}" textprefix="$t" text="{{{C_LabelTitle}}}" />

    <label id="Label_Description" pos="{{{C_FrameTextMargin}}} {{{-TitleY - C_FrameTextMargin}}}" halign="left" valign="top" size="{{{LabelWidth}}}" textfont="{{{SubFont}}}" textsize="1" textcolor="{{{C_ForegroundColor}}}" text="{{{C_LabelDescription}}}" autonewline="1" />

    <quad id="Quad_Background0" size="{{{C_FrameWidth}}} 18" bgcolor="{{{C_BackgroundColor}}}" opacity="{{{C_BackgroundOpacity}}}" />
    <quad id="Quad_Background1" size="{{{C_FrameWidth}}} 18" style="Bgs1" substyle="BgDialogBlur" opacity="0" />
  </frame>

  <frame id="Frame_ButtonOverrideContainer" pos="{{{HideOffset}}}">
    <label id="Label_IconEnabled" pos="{{{ButtonSize.Y / 2}}} {{{ButtonSize.Y / -2}}}" halign="center" valign="center2" textsize="1.75" textcolor="FFF" text="" />
    <label id="Label_IconDisabled" pos="{{{ButtonSize.Y / 2}}} {{{ButtonSize.Y / -2}}}" halign="center" valign="center2" textsize="1.75" textcolor="FFF" text="" />

    {{{DesignButtons::CreateButton("ButtonOverride", ButtonSize, C_LabelButton)}}}
  </frame>

  <frame id="Frame_BottomDecorLine" pos="{{{HideOffset}}}">
    <quad pos="0 0" size="{{{C_FrameWidth}}} 0.25" bgcolor="FFF" />
    <quad pos="{{{C_FrameWidth}}} 0" halign="right" size="15 0.75" bgcolor="FFF" />
  </frame>
</frame>

<script><!--

{{{MouseExt::InjectDeclares()}}}
{{{DesignButtons::InjectFunctions()}}}

Void SetButtonHover(CMlFrame _Frame, Boolean _IsHovered) {
  Page.GetClassChildren("Hover", _Frame, True);

  declare Opacity = 0.0;

  if (_IsHovered) {
    Opacity = 1.0;
  }

  foreach (Control in Page.GetClassChildren_Result) {
    AnimMgr.Flush(Control);
    AnimMgr.Add(Control, "<quad opacity='"^Opacity^"' />", 125, CAnimManager::EAnimManagerEasing::CircOut);
  }
}

Void SetVisibility(Boolean _IsVisible) {
  declare Frame_ClipContainer <=> (Page.GetFirstChild("Frame_ClipContainer") as CMlFrame);
  declare Frame_MoveContainer <=> (Page.GetFirstChild("Frame_MoveContainer") as CMlFrame);

  declare PosX = {{{HideOffset}}};
  declare BaseDuration = 500;
  declare Step = BaseDuration / 10;
  declare Easing = CAnimManager::EAnimManagerEasing::CubicIn;

  if (_IsVisible) {
    PosX = 0.0;
    Easing = CAnimManager::EAnimManagerEasing::CubicOut;
  }

  foreach (I => Control in Frame_MoveContainer.Controls) {
    declare PosY = Control.RelativePosition_V3.Y;
    declare Scale = I * Step;
    declare StartTime = Now;
    declare Duration = BaseDuration;

    if (_IsVisible) {
      StartTime += Scale;
      Duration -= Scale;
    } else {
      Duration += Scale;
    }

    AnimMgr.Flush(Control);
    AnimMgr.Add(Control, "<frame pos='"^PosX^" "^PosY^"' />", StartTime, Duration, Easing);
  }
}

Void SetIconEnabled(Boolean _IsEnabled) {
  declare Label_IconEnabled <=> (Page.GetFirstChild("Label_IconEnabled") as CMlLabel);
  declare Label_IconDisabled <=> (Page.GetFirstChild("Label_IconDisabled") as CMlLabel);

  Label_IconEnabled.Visible = _IsEnabled;
  Label_IconDisabled.Visible = !_IsEnabled;
}

main() {
  declare Label_Description <=> (Page.GetFirstChild("Label_Description") as CMlLabel);
  declare Quad_Background0 <=> (Page.GetFirstChild("Quad_Background0") as CMlQuad);
  declare Quad_Background1 <=> (Page.GetFirstChild("Quad_Background1") as CMlQuad);
  declare Frame_ButtonOverrideContainer <=> (Page.GetFirstChild("Frame_ButtonOverrideContainer") as CMlFrame);
  declare Frame_ButtonOverride <=> (Page.GetFirstChild("Frame_ButtonOverride") as CMlFrame);
  declare Frame_BottomDecorLine <=> (Page.GetFirstChild("Frame_BottomDecorLine") as CMlFrame);

  declare Height = Label_Description.ComputeHeight(Label_Description.Value);

  Quad_Background0.Size.Y = -Label_Description.RelativePosition_V3.Y + Height + {{{C_FrameTextMargin}}};
  Quad_Background1.Size.Y = Quad_Background0.Size.Y;

  Frame_ButtonOverrideContainer.RelativePosition_V3.Y = Quad_Background0.RelativePosition_V3.Y - Quad_Background0.Size.Y - {{{C_ButtonSpacingGap}}};
  Frame_BottomDecorLine.RelativePosition_V3.Y = Frame_ButtonOverrideContainer.RelativePosition_V3.Y - {{{ButtonSize.Y}}} - {{{C_ButtonSpacingGap}}};

  declare IsVisible = False;

  declare PrevVisibility = False;

  SetVisibility(IsVisible);

  while (True) {
    foreach (Event in PendingEvents) {
      if (Event.Type == CMlScriptEvent::Type::PluginCustomEvent) {
        switch (Event.CustomEventType) {
          case "{{{C_CustomEventType_SetIsVisible}}}": {
            if (Event.CustomEventData.count < 1) continue;

            IsVisible = Event.CustomEventData[0] == "True";
          }

          case "{{{C_CustomEventType_SetPreferCustom}}}": {
            if (!IsVisible || Event.CustomEventData.count < 1) continue;

            declare PrefersCustom = Event.CustomEventData[0] == "True";

            Button_SetFilled(Frame_ButtonOverride, PrefersCustom);
            SetIconEnabled(PrefersCustom);
          }
        }
      }

      else if (Event.Type == CMlScriptEvent::Type::MouseOut) {
        if (Event.ControlId != "Quad_ButtonOverride") continue;

        SetButtonHover(Event.Control.Parent, False);
      }
    }

    declare IsActuallyVisible = PageIsVisible && IsVisible;

    if (PrevVisibility != IsActuallyVisible) {
      SetVisibility(IsActuallyVisible);

      PrevVisibility = IsActuallyVisible;
    }

    if (!IsActuallyVisible) {
      yield;
      continue;
    }

    {{{MouseExt::InjectEvents(MouseExt::K_Options {
      MouseUp = """
        if (EventControl.ControlId == "Quad_ButtonOverride") {
          SendCustomEvent("{{{C_CustomEventType_TogglePreferCustom}}}", []);
        }
      """
    })}}}

    foreach (Event in PendingEvents) {
      if (Event.Type == CMlScriptEvent::Type::MouseClick) {
        if (Event.Control.HasClass("Focus")) {
          Audio.PlaySoundEvent(CAudioManager::ELibSound::Valid, 1, 0.0);
        }
      }

      else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
        if (Event.Control.HasClass("Focus")) {
          Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.0);
        }

        if (Event.ControlId != "Quad_ButtonOverride") continue;

        SetButtonHover(Event.Control.Parent, True);
      }
    }

    yield;
  }
}

--></script>
</manialink>
  """;
}
