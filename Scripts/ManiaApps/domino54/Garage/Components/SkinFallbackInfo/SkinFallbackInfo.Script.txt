#Const Version      "2025-04-22"
#Const ScriptName   "ManiaApps/domino54/Garage/Components/SkinFallbackInfo/SkinFallbackInfo.Script.txt"

#Include "TextLib" as TL
#Include "ManiaApps/Nadeo/Layers.Script.txt" as Layers
#Include "ManiaApps/domino54/Garage/Components/SkinFallbackInfo/LB_SkinFallbackInfo.Script.txt" as LB

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_ManialinkLayerId "L_SkinFallbackInfo"

#Const LB::C_CustomEventType_TogglePreferCustom   as C_CustomEventType_TogglePreferCustom
#Const LB::C_CustomEventType_SetPreferCustom      as C_CustomEventType_SetPreferCustom
#Const LB::C_CustomEventType_SetIsVisible         as C_CustomEventType_SetIsVisible

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void SetButtonVisibility(Boolean _IsVisible) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);

  LayerCustomEvent(Layer, C_CustomEventType_SetIsVisible, [TL::ToText(_IsVisible)]);
}

Void SetPreferCustom(Boolean _PreferCustom) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);

  LayerCustomEvent(Layer, C_CustomEventType_SetPreferCustom, [TL::ToText(_PreferCustom)]);
}

Void Unload() {
  Layers::Destroy(C_ManialinkLayerId);
}

Void Load() {
  Unload();

  Layers::Create(C_ManialinkLayerId, LB::Build());
  Layers::SetVisibility(C_ManialinkLayerId, False);
}

Void Show() {
  Layers::SetVisibility(C_ManialinkLayerId, True);
}

Void Hide() {
  Layers::SetVisibility(C_ManialinkLayerId, False);
}
