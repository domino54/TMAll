#Const Version      "2025-04-22"
#Const ScriptName   "ManiaApps/domino54/Garage/Components/ModelTabs/ModelTabs.Script.txt"

#Include "ManiaApps/Nadeo/Layers.Script.txt" as Layers
#Include "ManiaApps/domino54/Garage/Components/ModelTabs/LB_ModelTabs.Script.txt" as LB

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_ManialinkLayerId "L_ModelTabs"

#Const LB::C_CustomEventType_ActiveTab    as C_CustomEventType_ActiveTab
#Const LB::C_CustomEventType_SetHeader    as C_CustomEventType_SetHeader
#Const LB::C_CustomEventType_SetColor     as C_CustomEventType_SetColor
#Const LB::C_CustomEventType_UpdateTabs   as C_CustomEventType_UpdateTabs
#Const LB::C_CustomEventType_SetTabId     as C_CustomEventType_SetTabId

// ---------------------------------- //
// Structs
// ---------------------------------- //

#Struct K_MenuTab {
  Text Id;
  Text Label;
  Text IconUrl;
}

// ---------------------------------- //
// Private functions
// ---------------------------------- //

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void SetHeaderText(Text _Value) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);
  declare Text MenuTabsHeaderText for Layer.LocalPage;

  MenuTabsHeaderText = _Value;
  LayerCustomEvent(Layer, C_CustomEventType_SetHeader, []);
}

Void SetColor(Vec3 _Color) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);
  declare Vec3 MenuTabsLineColor for Layer.LocalPage;

  MenuTabsLineColor = _Color;
  LayerCustomEvent(Layer, C_CustomEventType_SetColor, []);
}

Void SetTabsData(K_MenuTab[] _TabsData) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);

  declare K_MenuTab[] MenuTabsData for Layer.LocalPage;

  MenuTabsData = _TabsData;

  LayerCustomEvent(Layer, C_CustomEventType_UpdateTabs, []);
}

Void SetActiveTabId(Text _ActiveTabId) {
  declare Layer <=> Layers::Get(C_ManialinkLayerId);

  LayerCustomEvent(Layer, C_CustomEventType_SetTabId, [_ActiveTabId]);
}

Void Unload() {
  Layers::Destroy(C_ManialinkLayerId);
}

Void Load() {
  Unload();

  Layers::Create(C_ManialinkLayerId, LB::Build());
  Layers::SetVisibility(C_ManialinkLayerId, False);
}

Void Show() {
  Layers::SetVisibility(C_ManialinkLayerId, True);
}

Void Hide() {
  Layers::SetVisibility(C_ManialinkLayerId, False);
}
