#Const Version      "2024-12-22"
#Const ScriptName   "ManiaApps/domino54/Garage/Components/GarageSplashScreen/GarageSplashScreen.Script.txt"

#Include "ManiaApps/Nadeo/Layers.Script.txt" as Layers
#Include "ManiaApps/domino54/Garage/Components/GarageSplashScreen/LB_GarageSplashScreen.Script.txt" as LB_GarageSplashScreen

// ---------------------------------- //
// Constants
// ---------------------------------- //

#Const C_ManialinkLayerId  "L_GarageSplashScreen"

// ---------------------------------- //
// Public functions
// ---------------------------------- //

Void SetVisibility(Boolean _IsVisible) {
  Layers::SetVisibility(C_ManialinkLayerId, _IsVisible);
}

Void Unload() {
  Layers::Destroy(C_ManialinkLayerId);
}

Void Load() {
  Unload();

  Layers::Create(C_ManialinkLayerId, LB_GarageSplashScreen::Build());
  Layers::SetVisibility(C_ManialinkLayerId, False);
}

Void Show() {
  Layers::SetVisibility(C_ManialinkLayerId, True);
}

Void Hide() {
  Layers::SetVisibility(C_ManialinkLayerId, False);
}
