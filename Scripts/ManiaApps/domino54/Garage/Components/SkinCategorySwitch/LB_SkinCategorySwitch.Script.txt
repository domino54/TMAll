#Const Version      "2025-08-01"
#Const ScriptName   "ManiaApps/domino54/Garage/Components/SkinCategorySwitch/LB_SkinCategorySwitch.Script.txt"

#Include "Libs/domino54/MouseExt.Script.txt" as MouseExt
#Include "ManiaApps/domino54/Horizon/Components/HorizonButtonToggle.Script.txt" as HorizonButtonToggle

#Const C_CustomEventType_UpdateCategories "category_switch:set_category"

Text Build(Integer _Budget) {
  return """
<manialink version="3" name="L_SkinCategorySwitch">

{{{HorizonButtonToggle::CreateModel("FrameModel_CategoryToggle", _Budget)}}}

<frame pos="-150 -52.5">
  <frameinstance modelid="FrameModel_CategoryToggle" id="Frame_CategoryToggle" />
</frame>

<script><!--

#Include "TextLib" as TL

{{{MouseExt::InjectDeclares()}}}
{{{HorizonButtonToggle::InjectFunctions()}}}

main() {
  declare Frame_CategoryToggle <=> (Page.GetFirstChild("Frame_CategoryToggle") as CMlFrame);

  declare Text[] CategoryNames for Page;
  declare Integer ActiveIndex for Page;

  HorizonButtonToggle_SetOptions(Frame_CategoryToggle, CategoryNames, -3.0);
  HorizonButtonToggle_SetIndex(Frame_CategoryToggle, ActiveIndex);

  while (True) {
    if (!PageIsVisible) {
      yield;
      continue;
    }

    {{{MouseExt::InjectEvents(MouseExt::K_Options {
      MouseUp = """
        if (EventControl.ControlId == "Quad_ButtonToggleTrigger") {
          declare Index = EventControl.Parent.Parent.Controls.keyof(EventControl.Parent);

          HorizonButtonToggle_SetIndex(Frame_CategoryToggle, Index);

          SendCustomEvent("{{{C_CustomEventType_UpdateCategories}}}", [TL::ToText(Index)]);
        }
      """
    })}}}

    foreach (Event in PendingEvents) {
      if (Event.Type == CMlScriptEvent::Type::MouseClick) {
        if (Event.Control.HasClass("Focus")) {
          Audio.PlaySoundEvent(CAudioManager::ELibSound::Valid, 1, 0.0);
        }
      }
      else if (Event.Type == CMlScriptEvent::Type::MouseOver) {
        if (Event.Control.HasClass("Focus")) {
          Audio.PlaySoundEvent(CAudioManager::ELibSound::Focus, 1, 0.0);
        }
      }
    }

    yield;
  }
}

--></script>
</manialink>
  """;
}
