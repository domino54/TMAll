// ---------------------------------- //
// Init server
// ---------------------------------- //
***MB_InitServer***
***
ClientManiaAppUrl = C_ManiaAppUrl;
MB_SetDefaultUIVisibility(False);
HideOpponents = True;
Replay_AutoStart = False;

declare Text PrevPreviewModelId;
declare Text PrevPreviewSkinId;
declare Integer LastDespawnTime;

// Spectating oneself prevents controlling of the car.
UIManager.UIAll.ForceSpectator = True;
UIManager.UIAll.SpectatorForcedTarget = Player.Id;
***

// ---------------------------------- //
// Init map
// ---------------------------------- //
***MB_InitMap***
***

UIManager.UIAll.UISequence_CustomMTClip = "HLoff_C_RFX";
UIManager.UIAll.UISequence_CustomMTRefTime = Now - 2500 + 250 - 75; // Fade delay
***

// ---------------------------------- //
// Yield
// ---------------------------------- //
***MB_Yield***
***
HttpClient::Yield();

foreach (Event in UIManager.PendingEvents) {
  if (Event.Type != CUIConfigEvent::EType::OnLayerCustomEvent) continue;

  if (Event.CustomEventType == Garage::C_CustomEventType_StopServer) {
    MB_Private_RunServer = False;
  }
}
***

// ---------------------------------- //
// Play loop
// ---------------------------------- //
***MB_PlayLoop***
***
foreach (Event in PendingEvents) {
  // Ignore all game events.
  Discard(Event);
}

// This will be changed by the UI layer.
declare PreviewModelId = Garage::GetPreviewModelId(LocalUser);
declare PreviewSkinId = Garage::GetPreviewSkinId(LocalUser);

// Why like this instead of events? Because it's more reliable.
if ((PrevPreviewModelId != PreviewModelId || PrevPreviewSkinId != PreviewSkinId) && Now >= LastDespawnTime + C_RespawnCooldown) {
  PrevPreviewModelId = PreviewModelId;
  PrevPreviewSkinId = PreviewSkinId;
  LastDespawnTime = Now;

  UnspawnPlayer(Player);
}

// Spawn player with the target model.
if (!Player.IsSpawned) {
  declare PreviewModelId = Garage::GetPreviewModelId(LocalUser);
  declare PreviewSkinId = Garage::GetPreviewSkinId(LocalUser);

  PlayerModels_Server::AssignModel(Player, PreviewModelId, PreviewSkinId, True);

  SpawnPlayer(Player, 0, Now);
  Garage::HideBackground(LocalUser);
}
***

Void MB_Yield() {
  yield;

  +++MB_Yield+++
}

Void MB_Private_LoadMap() {
  if (!MapLoaded) RequestLoadMap();
  while (!MapLoaded) MB_Yield();
}

Void MB_Private_UnloadMap() {
  if (MapLoaded) RequestUnloadMap();
  while (MapLoaded) MB_Yield();
}

Boolean MB_Private_ServerIsRunning() {
  return !ServerShutdownRequested && MB_Private_RunServer;
}

Boolean MB_Private_MatchIsRunning() {
  return MB_Private_ServerIsRunning() && !MatchEndRequested;
}

Void MB_Private_ItemList() {
  ItemList_Begin();
  +++MB_ItemList+++
  ItemList_End();
}

// ---------------------------------- //
// Script entry point
// ---------------------------------- //
main() {
  MB_Private_RunServer = True;

  wait(Players.count >= 1);

  declare Player <=> Players[0];
  declare LocalUser <=> Users[0];

  while (MB_Private_ServerIsRunning()) {
    +++MB_LoadLibraries+++
    +++MB_LogVersions+++

    +++MB_InitServer+++
    +++MB_StartServer+++

    MB_Private_ItemList();
    MB_Private_LoadMap();

    declare BarrierId = Synchro_AddBarrier();

    while (!Synchro_BarrierReached(BarrierId)) {
      MB_Yield();
    }

    +++MB_InitMap+++
    +++MB_StartMap+++

    while (MB_Private_MatchIsRunning()) {
      +++MB_PlayLoop+++

      // Have to yield at the end, to get updates as quickly as possible.
      MB_Yield();
    }

    +++MB_EndMap+++

    MB_Private_UnloadMap();
  }

  +++MB_EndServer+++
  +++MB_UnloadLibraries+++
}
