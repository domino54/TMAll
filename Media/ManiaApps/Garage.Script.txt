#RequireContext CManiaApp

#Extends "ManiaApps/domino54/Base/AppBase.Script.txt"

#Const Version    "2024-10-30"
#Const ScriptName "ManiaApps/Garage.Script.txt"

#Include "ManiaApps/domino54/Garage/Core/GarageAPI.Script.txt" as GarageAPI
#Include "ManiaApps/domino54/Garage/Core/GarageController.Script.txt" as GarageController

#Setting S_ScriptEnvironment "development"

// ---------------------------------- //
// Init Garage
// ---------------------------------- //
***AB_InitApp***
***
if (GarageAPI::IsPlayground()) {
  // Mutes the car engine.
  Audio.LimitSceneSoundVolumedB = -40.0;

  // Curiously does not pause the game in solo mode, yet enables menu controls.
  // Only used in-game, because it breaks the scroll inside manialinks???
  EnableMenuNavigationInputs = True;
}

GarageAPI::Load();
GarageController::Init();

while (GarageAPI::IsLoading()) {
  AB_Yield();
}

GarageAPI::SetModelId(GarageAPI::GetInitialModelId());
GarageController::Load();

while (GarageAPI::IsPlayground() && !GarageAPI::IsServerReady()) {
  AB_Yield();
}

// Ugly way to get the UI to draw later.
if (GarageAPI::IsPlayground()) {
  AB_Sleep(50);
}

GarageController::Show();
***

// ---------------------------------- //
// Garage Loop
// ---------------------------------- //
***AB_AppLoop***
***
GarageController::Loop();
***

// ---------------------------------- //
// Stop Garage
// ---------------------------------- //
***AB_EndApp***
***
GarageController::Unload();
***
